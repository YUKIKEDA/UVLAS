cmake_minimum_required(VERSION 3.25)
project(UVLASIOBindingSpike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(UVLAS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${UVLAS_ROOT}/cmake/CompilerPolicy.cmake")
find_package(Python 3.12 REQUIRED COMPONENTS Interpreter Development.Module)
set(PYBIND11_FINDPYTHON ON)
add_subdirectory("${UVLAS_ROOT}/build/numerical-deps/pybind11-3.1.0" "${CMAKE_CURRENT_BINARY_DIR}/pybind11" EXCLUDE_FROM_ALL)
pybind11_add_module(_uvlas_design_binding binding.cpp)
target_include_directories(_uvlas_design_binding SYSTEM PRIVATE "${UVLAS_ROOT}/build/numerical-deps/json-3.12.0/single_include")
if(MSVC)
  target_compile_options(_uvlas_design_binding PRIVATE /utf-8 /fp:strict)
else()
  target_compile_options(_uvlas_design_binding PRIVATE -ffp-contract=off -fno-fast-math)
endif()
enable_testing()
add_test(NAME ownership_and_json COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_binding.py" "$<TARGET_FILE_DIR:_uvlas_design_binding>")
