constexpr core::Index use_chart_case = 2;
constexpr std::size_t failing_id = 3;

std::string frozen_interface(core::Index file) {
    const auto path = std::string("r4-interface-0") + char('0' + file) + "/case.json";
    std::ifstream stream(path, std::ios::binary);
    REQUIRE(stream);
    return {std::istreambuf_iterator<char>{stream}, std::istreambuf_iterator<char>{}};
}

cc::CasePgChartLimits chart_limits(core::Index right_n) {
    auto arithmetic = query_limits();
    arithmetic.maximum_scalar_operations = 1'000'000'000;
    arithmetic.maximum_partition_visits = 300'000'000;
    auto chart = arithmetic;
    chart.algebra.maximum_value_bits = 65536;
    chart.algebra.maximum_work_bits = 262144;
    // The nonmatching layout uses the existing frozen R4 chart capacities.
    const bool nonmatching = right_n == 2;
    const auto trace_terms = nonmatching ? 2'000'000U : 120'000U;
    const PgGlobalConstraintLimits global{chart, 2, 2, nonmatching ? 4096U : 500U, trace_terms};
    const PgSparseChartLimits sparse{chart, nonmatching ? 384U : 300U, nonmatching ? 1024U : 480U,
                                     nonmatching ? 100'000U : 70'000U};
    return {case_pg_reference_limits(),
            {profile, steps, 128, {}},
            {768, 704},
            {arithmetic, steps},
            {768, 128},
            {profile, steps, {}},
            nonmatching ? chart : arithmetic,
            {motion_exact_limits(), 10'000, cc::case_motion_frame_interval_scratch_bytes},
            {chart, trace_terms},
            global,
            sparse,
            100'000};
}

core::Result<cc::CasePgChart> prepare_fixture(std::string_view wire, core::Index right_n,
                                              core::Workspace& workspace,
                                              core::ResourcePlan& plan) {
    const auto source = checked_case_wire(wire);
    const auto limits = chart_limits(right_n);
    Fixture f;
    auto selection = core::ResourcePlan{profile};
    const auto nodes = finite_rule(f);
    std::array<std::vector<ExactRational>, use_chart_case> weights;
    std::array<std::vector<PgDiscreteVolumePoint>, use_chart_case> points;
    for (std::size_t i = 0; i < points.size(); ++i) {
        const auto n = i == 0 ? 1 : right_n;
        auto geometry = take(
            cc::prepare_case_pg_reference(source, i + 1, limits.reference, workspace, selection));
        const auto owners = take(geometry.source());
        auto rank = take(PgRankPreparation::prepare(owners.dictionary, owners.partition,
                                                    limits.rank, workspace, selection));
        auto basis = take(PgReferenceBasis::prepare(std::move(geometry), std::move(rank),
                                                    limits.rank, workspace, selection));
        points[i] = select_rule(f, basis, n, nodes, weights[i], selection);
    }
    const std::array rules{cc::CasePgRuleBinding{1, points[0]},
                           cc::CasePgRuleBinding{2, points[1]}};
    core::ResourcePlan empty{0};
    const std::array unexpected{cc::CasePgRuleBinding{failing_id, points[0]}, rules[1]};
    check_failure(cc::prepare_case_pg_chart(source, unexpected, limits, workspace, empty),
                  core::FailureCode::InvalidInput);
    check_failure(cc::prepare_case_pg_chart(source, rules, limits, workspace, empty),
                  core::FailureCode::ResourceExceeded);
    check_failure(
        cc::prepare_case_pg_chart(source, rules, limits, workspace, empty, requested_stop_token()),
        core::FailureCode::Cancelled);
    CHECK(empty.allocated_bytes() == 0);
    return cc::prepare_case_pg_chart(source, rules, limits, workspace, plan);
}

Q work(std::span<const PgCartesianExactPosition> a, std::span<const PgCartesianExactPosition> b) {
    REQUIRE(a.size() == b.size());
    Q result{};
    for (std::size_t i = 0; i < a.size(); ++i)
        for (std::size_t j = 0; j < a[i].size(); ++j)
            result += q(a[i][j]) * q(b[i][j]);
    return result;
}

void verify_rows(const cc::CasePgChart& owner, core::Index n, core::Workspace& workspace,
                 core::ResourcePlan& plan) {
    const auto limits = chart_limits(n);
    const auto& chart = owner.chart();
    const auto identity = take(chart.identity());
    const auto& original = chart.original();
    const auto layout = original.layout();
    const auto sources = original.sources();
    REQUIRE(owner.patch_count() == 2);
    REQUIRE(layout.size() == 2);
    REQUIRE(sources.size() == 2);
    CHECK(sources[0].kind == PgTraceKind::Connection);
    CHECK(sources[1].kind == PgTraceKind::Support);
    CHECK(sources[0].id == PgConstraintId{1});
    CHECK(sources[1].id == PgConstraintId{1});
    CHECK(sources[0].overlaps == 2 * n);
    CHECK(sources[1].overlaps == 2);
    CHECK(sources[0].left == numerics::PatchId{1});
    CHECK(sources[0].right == numerics::PatchId{2});
    CHECK(sources[1].left == numerics::PatchId{1});
    REQUIRE_FALSE(sources[1].right.has_value());

    Fixture fixture;
    const auto size = static_cast<std::size_t>(take(original.info()).columns);
    std::vector<PgCartesianExactPosition> x0(size), force(size);
    for (std::size_t p = 0; p < layout.size(); ++p) {
        const auto patch = take(owner.patch(static_cast<core::Index>(p)));
        const auto info = take(patch->info());
        REQUIRE(layout[p].basis == info.identity);
        REQUIRE(layout[p].id.value() == p + 1);
        REQUIRE(patch->fixed_rule().has_value());
        const auto positions = patch->reference_coordinates();
        REQUIRE(positions.size() == static_cast<std::size_t>(layout[p].count));
        for (std::size_t i = 0; i < positions.size(); ++i)
            for (std::size_t d = 0; d < positions[i].size(); ++d)
                x0[static_cast<std::size_t>(layout[p].first) + i][d] =
                    owned(fixture, q(positions[i][d]));
    }
    const auto residual =
        take(original.residual({layout, x0}, limits.chart.arithmetic, workspace, plan));
    REQUIRE(residual.values().size() == original.original_rows().size());
    for (const auto& at : residual.values())
        for (const auto& component : at)
            CHECK(component.sign() == 0);
    for (std::size_t i = 0; i < force.size(); ++i)
        force[i][i % 3] = owned(fixture, Q{static_cast<unsigned>(i % 7 + 1), 17});
    std::vector<PgCartesianExactPosition> free(chart.free_columns().size());
    for (std::size_t i = 0; i < free.size(); ++i)
        for (std::size_t d = 0; d < free[i].size(); ++d)
            free[i][d] =
                owned(fixture, q(x0[static_cast<std::size_t>(chart.free_columns()[i])][d]));
    const auto lifted =
        take(chart.lift({identity, free}, limits.chart.arithmetic, workspace, plan));
    for (std::size_t i = 0; i < lifted.values().size(); ++i)
        for (std::size_t d = 0; d < lifted.values()[i].size(); ++d)
            CHECK(q(lifted.values()[i][d]) == q(x0[i][d]));

    for (std::size_t i = 0; i < free.size(); ++i)
        free[i][i % 3] = owned(fixture, -Q{static_cast<unsigned>(i % 5 + 1), 19});
    const auto tangent =
        take(chart.tangent({identity, free}, limits.chart.arithmetic, workspace, plan));
    const auto dual =
        take(chart.transpose({identity, force}, limits.chart.arithmetic, workspace, plan));
    CHECK(work(tangent.values(), force) == work(free, dual.independent_force.values()));
    const auto support_force = take(original.transpose(dual.original_multipliers.values(),
                                                       limits.chart.arithmetic, workspace, plan));
    std::vector<std::array<Q, 3>> reaction(size);
    for (std::size_t i = 0; i < original.original_rows().size(); ++i) {
        const auto row = take(original.row(static_cast<core::Index>(i)));
        for (const auto& term : row)
            for (std::size_t d = 0; d < 3; ++d)
                reaction[static_cast<std::size_t>(term.column)][d] +=
                    q(term.coefficient) * q(dual.original_multipliers.values()[i][d]);
    }
    for (std::size_t i = 0; i < size; ++i)
        for (std::size_t d = 0; d < 3; ++d)
            CHECK(q(support_force.values()[i][d]) == reaction[i][d]);
    const auto measured = take(chart.preparation());
    std::clog << "CASE_PG_ACCEPT," << n << ',' << size << ',' << residual.values().size() << ','
              << chart.pivots().size() << ',' << plan.allocated_bytes() << ','
              << measured.original_nonzeros << ',' << measured.lower_terms << ','
              << measured.upper_terms << ',' << measured.factor_term_capacity << '\n'
              << std::flush;
}
} // namespace

TEST_CASE("The frozen owned Cases prepare the exact two-panel PG sparse chart",
          "[.frozen-vpg-chart][V-PG][SH-D05][CN-D01][N4-04][verification]") {
    for (core::Index i = 0; i < 2; ++i) {
        DYNAMIC_SECTION("interface " << i) {
            core::Workspace workspace{core::PlannedBuffer<std::byte>{}};
            core::ResourcePlan plan{profile};
            const auto ready = prepare_fixture(frozen_interface(i), i + 1, workspace, plan);
            if (!ready)
                UNSCOPED_INFO("Case chart refusal: code "
                              << static_cast<unsigned>(ready.error().code) << ", operation "
                              << static_cast<unsigned>(ready.error().operation)
                              << ", cumulative bytes " << plan.allocated_bytes());
            REQUIRE(ready);
            verify_rows(*ready, i + 1, workspace, plan);
        }
    }
}

TEST_CASE("A reversed named Case edge rejects a same-orientation declaration",
          "[.frozen-vpg-chart-negative][V-PG][CN-D01][verification][negative]") {
    auto wire = nlohmann::json::parse(frozen_interface(0));
    for (auto& entry : wire.at("model").at("joints").at(0).at("mapping"))
        entry.at("orientation") = "same";
    core::Workspace workspace{core::PlannedBuffer<std::byte>{}};
    core::ResourcePlan plan{profile};
    check_failure(prepare_fixture(wire.dump(), 1, workspace, plan),
                  core::FailureCode::GeometryInvalid);
}

TEST_CASE("The fixed Case support rejects a reflected declared motion frame",
          "[.frozen-vpg-chart-negative][V-PG][CN-D01][verification][negative]") {
    auto wire = nlohmann::json::parse(frozen_interface(0));
    wire.at("model").at("supports").at(0).at("motion").at("origin").at("rotation").at(2).at(2) = -1;
    core::Workspace workspace{core::PlannedBuffer<std::byte>{}};
    core::ResourcePlan plan{profile};
    check_failure(prepare_fixture(wire.dump(), 1, workspace, plan),
                  core::FailureCode::GeometryInvalid);
}
