struct Contact {
    std::size_t left, right, fixed_axis;
};

constexpr std::array contacts{Contact{0, 1, 0}, Contact{0, 2, 1}, Contact{2, 3, 0},
                              Contact{1, 3, 1}};
constexpr std::array patch_ids{numerics::PatchId{17}, numerics::PatchId{72}, numerics::PatchId{45},

constexpr std::array contact_ids{PgConstraintId{118}, PgConstraintId{315}, PgConstraintId{406},
                                 PgConstraintId{507}};

struct Phase {
    std::uint64_t nanoseconds, charged_bytes;
    core::ProcessMemory memory;
};

Phase finish(Clock::time_point begin, const core::ResourcePlan& plan, std::uint64_t bytes_before) {
    const auto duration =
        std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now() - begin);
    return {static_cast<std::uint64_t>(duration.count()), plan.allocated_bytes() - bytes_before,
            take(core::sample_process_memory(core::Operation::PrepareModel))};
}

void write_phase(std::string_view name, const Phase& phase) {
    std::cout << ",\"" << name << "\":{\"nanoseconds\":" << phase.nanoseconds
              << ",\"charged_bytes\":" << phase.charged_bytes
              << ",\"resident_bytes\":" << phase.memory.resident_bytes
              << ",\"lifetime_peak_resident_bytes\":" << phase.memory.peak_resident_bytes << '}';
}

PgPreparedPatch build_patch(Fixture& f, const PgReferenceFrame& frame, core::Workspace& ws,
                            core::ResourcePlan& plan) {
    const auto limits = query_limits();
    auto gram =
        take(PgFlatGram::prepare(make_basis(f, frame, ws, plan), 256, {limits, steps}, ws, plan));
    auto accepted =
        take(std::move(gram).generate_and_certify({320, 246}, {profile, steps, {}}, ws, plan));
    return take(PgPreparedPatch::prepare(std::move(accepted), limits, ws, plan));
}

struct EdgeRange {
    core::Index edge;
    ExactRational first, last;
    bool forward;
};

std::vector<EdgeRange> ranges(Fixture& f, const PgPreparedPatch& patch, std::size_t axis,
                              const Q& fixed) {
    const auto& part = take(take(patch.conditioning().reference_basis())->source()).partition;
    const auto vertices = take(part.vertices());
    const auto edges = take(part.edges());
    std::vector<EdgeRange> result;
    for (std::size_t i = 0; i < edges.size(); ++i) {
        const auto& edge = edges[i];
        if (!edge.external)
            continue;
        const auto& a = vertices[static_cast<std::size_t>(edge.vertices[0])];
        const auto& b = vertices[static_cast<std::size_t>(edge.vertices[1])];
        if ((axis == 0 ? q(a.s) : q(a.t)) != fixed || (axis == 0 ? q(b.s) : q(b.t)) != fixed)
            continue;
        const auto x = axis == 0 ? q(a.t) : q(a.s);
        const auto y = axis == 0 ? q(b.t) : q(b.s);
        const auto lo = std::min(x, y);
        const auto hi = std::max(x, y);
        result.push_back({static_cast<core::Index>(i), rat(f, numerator(lo), denominator(lo)),
                          rat(f, numerator(hi), denominator(hi)), x < y});
    }
    std::ranges::sort(result, [](const auto& a, const auto& b) { return q(a.first) < q(b.first); });
    REQUIRE_FALSE(result.empty());
    REQUIRE(q(result.front().first) == 0);
    REQUIRE(q(result.back().last) == 1);
    for (std::size_t i = 1; i < result.size(); ++i)
        REQUIRE(result[i - 1].last == result[i].first);
    return result;
}

std::vector<PgTraceEdgeSpan> spans(const std::vector<EdgeRange>& ranges, const ExactRational& zero,
                                   const ExactRational& one) {
    std::vector<PgTraceEdgeSpan> result;
    result.reserve(ranges.size());
    for (const auto& r : ranges)
        result.push_back({r.edge, r.first, r.last, r.forward ? zero : one, r.forward ? one : zero,
                          std::nullopt});
    return result;
}

bool unequal_intervals(const std::vector<EdgeRange>& a, const std::vector<EdgeRange>& b) {
    return !std::ranges::equal(a, b, [](const EdgeRange& x, const EdgeRange& y) {
        return x.first == y.first && x.last == y.last;
    });
}

#ifdef UVLAS_R02_MEASURE_REUSE
Q work(std::span<const PgCartesianExactPosition> a, std::span<const PgCartesianExactPosition> b) {
    REQUIRE(a.size() == b.size());
    Q result{0};
    for (std::size_t i = 0; i < a.size(); ++i)
        for (std::size_t c = 0; c < 3; ++c)
            result += q(a[i][c]) * q(b[i][c]);
    return result;
}

std::vector<PgCartesianExactPosition> sample(Fixture& f, std::size_t count, std::size_t offset) {
    std::vector<PgCartesianExactPosition> result(count);
    for (std::size_t i = 0; i < count; ++i)
        for (std::size_t c = 0; c < 3; ++c)
            result[i][c] = f.r(std::to_string(static_cast<int>((i + 2 * c + offset) % 9) - 4));
    return result;
}

std::uint64_t external_forecast(const PgReferenceLimits& limits, std::uint64_t live_rationals,
                                core::ResourcePlan& plan) {
    detail::ReferenceArithmetic admission{limits, plan};
    const auto held = admission.external_integer_payload(2 * live_rationals + 1);
    constexpr std::uint64_t width = detail::IntegerBackend::limb_bits;
    const auto work_bits = (limits.algebra.maximum_work_bits / width - 2) * width;
    return take(detail::exact_rational_admission(limits.algebra, work_bits, held))
        .estimated_external_payload_bytes;
}

void measure_reuse(std::size_t patches, std::size_t connections, const PgSparseChart& chart,
                   const PgSparseChartPreparation& prepared, const PgGlobalConstraintInfo& info,
                   const Phase& preparation, Fixture& fixture, const PgReferenceLimits& limits,
                   core::Workspace& workspace, core::ResourcePlan& plan) {
    const auto columns = static_cast<std::size_t>(info.columns);
    const auto rows = static_cast<std::size_t>(info.rows);
    const auto independent = sample(fixture, chart.free_columns().size(), 1);
    const auto force = sample(fixture, columns, 3);
    const auto identity = take(chart.identity());
    REQUIRE(independent.size() + prepared.rank == columns);
    const auto chart_live = detail::reference_live_rationals + prepared.factor_term_capacity +
                            2 * prepared.work_row_capacity + prepared.pivot_capacity +
                            3 * (columns + prepared.pivot_capacity);
    const auto chart_estimate = external_forecast(limits, chart_live, plan);
    const auto lift_live = detail::reference_live_rationals + 3 * (columns + prepared.rank);
    const auto transposed_live = detail::reference_live_rationals + 3 * (2 * columns + rows);
    const auto lift_estimate = external_forecast(limits, lift_live, plan);
    const auto transposed_estimate = external_forecast(limits, transposed_live, plan);
    const auto before = take(core::sample_process_memory(core::Operation::PrepareModel));

    auto bytes = plan.allocated_bytes();
    auto start = Clock::now();
    const auto lifted = take(chart.lift({identity, independent}, limits, workspace, plan));
    const auto lift = finish(start, plan, bytes);
    REQUIRE(lifted.size() == columns);

    bytes = plan.allocated_bytes();
    start = Clock::now();
    const auto dual = take(chart.transpose({identity, force}, limits, workspace, plan));
    const auto transpose = finish(start, plan, bytes);
    REQUIRE(dual.independent_force.size() == independent.size());
    REQUIRE(dual.original_multipliers.size() == rows);
    REQUIRE(take(chart.identity()) == identity);

    const auto& original = chart.original();
    const auto residual =
        take(original.residual({original.layout(), lifted.values()}, limits, workspace, plan));
    REQUIRE(std::ranges::all_of(residual.values(), [](const PgCartesianExactPosition& row) {
        return std::ranges::all_of(row, [](const auto& x) { return x.sign() == 0; });
    }));
    const Q direct = work(lifted.values(), force);
    REQUIRE(direct != 0);
    REQUIRE(direct == work(independent, dual.independent_force.values()));

    std::cout << "R02-REUSE {\"patches\":" << patches << ",\"connections\":" << connections
              << ",\"rank\":" << prepared.rank << ",\"nonzeros\":" << info.nonzeros
              << ",\"lower_terms\":" << prepared.lower_terms
              << ",\"upper_terms\":" << prepared.upper_terms
              << ",\"resident_before_reuse\":" << before.resident_bytes
              << ",\"chart_live_rationals\":" << chart_live
              << ",\"chart_estimated_external_bytes\":" << chart_estimate
              << ",\"lift_live_rationals\":" << lift_live
              << ",\"lift_estimated_external_bytes\":" << lift_estimate
              << ",\"transpose_live_rationals\":" << transposed_live
              << ",\"transpose_estimated_external_bytes\":" << transposed_estimate
              << ",\"lift_temporary_triples\":" << prepared.rank
              << ",\"transpose_temporary_triples\":" << columns
              << ",\"lift_retained_triples\":" << lifted.size()
              << ",\"transpose_retained_triples\":"
              << dual.independent_force.size() + dual.original_multipliers.size();
    write_phase("chart", preparation);
    write_phase("lift", lift);
    write_phase("transpose", transpose);
    std::cout << "}\n";
}

void run(std::size_t patch_count, std::size_t contact_count) {
    const auto start_memory = take(core::sample_process_memory(core::Operation::PrepareModel));
    core::Workspace workspace{core::PlannedBuffer<std::byte>{}};
    core::ResourcePlan plan{profile};
    PgGlobalConstraintLimits limits{query_limits(), 4, 5, 4096, 150'000};
    limits.arithmetic.maximum_scalar_operations = 300'000'000;
    limits.arithmetic.maximum_partition_visits = 100'000'000;
    const PgTraceLimits trace_limits{limits.arithmetic, limits.maximum_nonzeros};
    PgSparseChartLimits chart_limits{limits.arithmetic, 512, 768, 65'000, true};
    chart_limits.arithmetic.algebra.maximum_value_bits = 65536;
    chart_limits.arithmetic.algebra.maximum_work_bits = 262144;

    auto begin = Clock::now();
    std::array fixtures{Fixture{1}, Fixture{2}, Fixture{1}, Fixture{1}};
    const auto zero = fixtures[0].r("0");
    const auto one = fixtures[0].r("1");
    std::array<ExactRational, 9> rotation{};
    for (const auto i : {0U, 4U, 8U})
        rotation[i] = fixtures[0].r("1");
    std::array<std::optional<PgPreparedPatch>, 4> patches;
    std::vector<PgConstraintPatchBinding> patch_bindings;
    patch_bindings.reserve(patch_count);
    for (std::size_t i = 0; i < patch_count; ++i) {
        const std::array<ExactRational, 3> origin{fixtures[0].r(i % 2U == 0 ? "0" : "1"),
                                                  fixtures[0].r(i / 2U == 0 ? "0" : "1"),
                                                  fixtures[0].r("0")};
        patches[i] = build_patch(fixtures[i], {rotation, origin, one}, workspace, plan);
        patch_bindings.push_back({patch_ids[i], *patches[i]});
    }
    const auto geometry_phase = finish(begin, plan, 0);

    auto bytes_before = plan.allocated_bytes();
    begin = Clock::now();
    std::array<std::optional<PgFullTrace>, 4> traces;
    std::vector<PgConstraintTraceBinding> bindings;
    bindings.reserve(contact_count + 1);
    std::array<std::array<std::size_t, 3>, 4> edge_counts{};
    std::size_t nonconforming = 0;
    for (std::size_t i = 0; i < contact_count; ++i) {
        const auto [left, right, axis] = contacts[i];
        const auto a = ranges(fixtures[left], *patches[left], axis, Q{1});
        const auto b = ranges(fixtures[right], *patches[right], axis, Q{0});
        if (unequal_intervals(a, b))
            ++nonconforming;
        traces[i] = take(PgFullTrace::prepare(*patches[left], spans(a, zero, one), *patches[right],
                                              spans(b, zero, one), trace_limits, workspace, plan));
        const auto t = take(traces[i]->info());
        edge_counts[i] = {a.size(), b.size(), t.overlaps};
        bindings.push_back({contact_ids[i], *traces[i], static_cast<core::Index>(left),
                            static_cast<core::Index>(right)});
    }
    REQUIRE(nonconforming > 0);
    const auto a_left = ranges(fixtures[0], *patches[0], 0, Q{0});
    const auto fixed_coords =
        std::vector<PgCartesianExactPosition>(patches[0]->reference_coordinates().size());
    const auto fixed = take(PgFullTrace::prepare_support(
        *patches[0], spans(a_left, zero, one), take(patches[0]->info()).identity, fixed_coords,
        trace_limits, workspace, plan));
    bindings.push_back({PgConstraintId{999}, fixed, 0, std::nullopt});
    const auto traces_phase = finish(begin, plan, bytes_before);

    const auto aggregate = [&] {
        return take(
            PgGlobalConstraints::prepare(patch_bindings, bindings, limits, workspace, plan));
    };

    bytes_before = plan.allocated_bytes();
    begin = Clock::now();
    auto original = aggregate();
    const auto aggregate_phase = finish(begin, plan, bytes_before);
    const auto original_info = take(original.info());
    const auto prepare = [&](PgGlobalConstraints& input) {
        return PgSparseChart::prepare(std::move(input), chart_limits, workspace, plan);
    };

    bytes_before = plan.allocated_bytes();
    begin = Clock::now();
    PgSparseChartPreparation measured;
    PgSparseChartIdentity identity;
    Phase chart_phase;
    {
        const auto chart = take(prepare(original));
        chart_phase = finish(begin, plan, bytes_before);
        measured = take(chart.preparation());
        identity = take(chart.identity());
        REQUIRE(measured.timing.has_value());
        REQUIRE(measured.original_nonzeros == static_cast<std::uint64_t>(original_info.nonzeros));
        REQUIRE(measured.rank > 0);
        REQUIRE(measured.upper_terms >= measured.rank);
        REQUIRE(measured.lower_terms + measured.upper_terms <= measured.factor_term_capacity);
#ifdef UVLAS_R02_MEASURE_REUSE
        measure_reuse(patch_count, contact_count, chart, measured, original_info, chart_phase,
                      fixtures[0], chart_limits.arithmetic, workspace, plan);
        return;
#endif
    }

    const auto exact_capacity = measured.lower_terms + measured.upper_terms;
    chart_limits.maximum_factor_terms = exact_capacity;
    {
        auto exact_original = aggregate();
        const auto exact = take(prepare(exact_original));
        REQUIRE(take(exact.identity()) == identity);
        const auto info = take(exact.preparation());
        REQUIRE(info.rank == measured.rank);
        REQUIRE(info.lower_terms == measured.lower_terms);
        REQUIRE(info.upper_terms == measured.upper_terms);
        REQUIRE(info.factor_term_capacity == exact_capacity);
    }
    chart_limits.maximum_factor_terms = exact_capacity - 1;
    auto short_original = aggregate();
    const auto short_bytes_before = plan.allocated_bytes();
    check_failure(prepare(short_original), core::FailureCode::ResourceExceeded);
    REQUIRE(take(short_original.info()).nonzeros == original_info.nonzeros);
    const auto short_bytes = plan.allocated_bytes() - short_bytes_before;

    std::cout << "R02 {\"patches\":" << patch_count << ",\"connections\":" << contact_count
              << ",\"cycle\":" << (contact_count == 4 ? "true" : "false")
              << ",\"nonconforming_contacts\":" << nonconforming << ",\"edge_counts\":[";
    for (std::size_t i = 0; i < contact_count; ++i) {
        const auto [a, b, common] = edge_counts[i];
        std::cout << (i == 0 ? "" : ",") << '[' << a << ',' << b << ',' << common << ']';
    }
    std::cout << "],\"rows\":" << original_info.rows << ",\"columns\":" << original_info.columns
              << ",\"nonzeros\":" << measured.original_nonzeros << ",\"rank\":" << measured.rank
              << ",\"lower_terms\":" << measured.lower_terms
              << ",\"upper_terms\":" << measured.upper_terms
              << ",\"pivot_capacity\":" << measured.pivot_capacity
              << ",\"work_row_capacity\":" << measured.work_row_capacity
              << ",\"factor_term_capacity\":" << measured.factor_term_capacity
              << ",\"internal_chart_nanoseconds\":" << measured.timing->elapsed_nanoseconds
              << ",\"exact_factor_capacity_accepted\":" << exact_capacity
              << ",\"one_less_factor_capacity_rejected\":" << exact_capacity - 1
              << ",\"one_less_charged_bytes\":" << short_bytes
              << ",\"initial_resident_bytes\":" << start_memory.resident_bytes;
    write_phase("geometry", geometry_phase);
    write_phase("traces", traces_phase);
    write_phase("aggregate", aggregate_phase);
    write_phase("chart", chart_phase);
    std::cout << "}\n";
}
} // namespace

TEST_CASE("R02 two patches") {
    run(2, 1);
}
TEST_CASE("R02 three patches") {
    run(3, 2);
}
TEST_CASE("R02 four patch loop") {
    run(4, 4);
