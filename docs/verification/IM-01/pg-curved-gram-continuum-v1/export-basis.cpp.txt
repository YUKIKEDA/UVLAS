#include "pg_conditioning_test_support.hpp"

#include <iostream>

using namespace uvlas;
using namespace uvlas::structure;
using namespace pg_conditioning_test;

TEST_CASE("Export the cut parabola basis for a continuum Gram oracle") {
    Fixture input;
    input.domain.clear();
    input.thickness.clear();
    input.heights.clear();
    const auto add = [&](std::string_view s, std::string_view tn, std::string_view td,
                         std::string_view hn, std::string_view hd) {
        input.domain.push_back({input.r(s), input.r(tn, td)});
        input.thickness.push_back({input.r(s), input.r(tn, td)});
        input.heights.push_back(input.r(hn, hd));
    };
    add("0", "0", "1", "1", "1");
    add("1", "1", "4", "3", "2");
    add("1", "7", "8", "3", "2");
    add("0", "1", "1", "1", "1");
    const auto zero = input.r("0"), one = input.r("1"), length = input.r("2");
    const auto quadratic = input.r("1", "16");
    const std::array<ExactInputTerm, 1> terms{{{{2, 0}, quadratic}}};
    const auto curve = take(ExactChartPolynomial::make(terms, exact_limits(), input.input_plan));
    input.graph = take(PgGraph::prepare({zero, one, zero, curve, 1, PgGraphConvention::Version1},
                                       {exact_limits(), 64, {}}, input.input_plan));
    core::ResourcePlan memory{profile};
    core::Workspace scratch{core::PlannedBuffer<std::byte>{}};
    const auto basis = make_basis(input, length, scratch, memory);
    const auto& partition = take(basis.source()).partition;
    const auto volumes = take(partition.volumes());
    const auto triangles = take(partition.triangles());
    const auto vertices = take(partition.vertices());
    const auto layers = take(partition.layers());
    const auto pieces = take(partition.pieces());
    for (std::size_t volume_index = 0; volume_index < volumes.size(); ++volume_index) {
        const auto& volume = volumes[volume_index];
        const auto& triangle = triangles[static_cast<std::size_t>(volume.triangle)];
        const auto cell = pieces[static_cast<std::size_t>(triangle.piece)].cell;
        const auto& layer = layers[static_cast<std::size_t>(volume.layer)];
        std::cout << "volume," << volume_index << ',' << cell.first << ',' << cell.second;
        for (const auto vertex : triangle.vertices) {
            const auto& point = vertices[static_cast<std::size_t>(vertex)];
            std::cout << ',' << q(point.s) << ',' << q(point.t);
        }
        std::cout << ',' << q(layer.lower) << ',' << q(layer.upper) << '\n';
    }
    const auto rank = take(basis.size());
    const auto raw = basis_terms(*take(basis.rank()), static_cast<std::size_t>(rank), memory);
    std::cout << "column,cell_s,cell_t,field,power_s,power_t,coefficient\n";
    for (std::size_t column = 0; column < raw.size(); ++column) {
        for (const auto& term : raw[column]) {
            const auto& row = term.row;
            std::cout << column << ',' << row.cell.first << ',' << row.cell.second << ','
                      << static_cast<unsigned>(row.field) << ',' << row.powers.first << ','
                      << row.powers.second << ',' << term.value << '\n';
        }
    }
    REQUIRE(rank == 130);
}
