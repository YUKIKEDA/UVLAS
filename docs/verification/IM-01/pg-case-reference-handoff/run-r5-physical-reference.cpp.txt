namespace {

using Quadratic = std::array<Q, 2>;

struct FieldJet {
    Decimal value;
    Vector gradient;
};

FieldJet independent_field(const Point& xy, const Decimal& depth, const Decimal& h,
                           std::size_t mode) {
    const auto& [s, t] = xy;
    const auto reference = physical_r5(xy, 2 * depth - 1, h);
    const Decimal fs = 2 * s / 5 + s * s / 10 + s * t * t / 20;
    const Decimal ft = t / 5 - 3 * t * t / 50 + s * s * t / 20;
    const Decimal fss = Decimal(2) / 5 + s / 5 + t * t / 20;
    const Decimal fst = s * t / 10;
    const Decimal ftt = Decimal(1) / 5 - 3 * t / 25 + s * s / 20;
    const Decimal g = 1 + fs * fs + ft * ft, g32 = g * sqrt(g);
    const Decimal gs = 2 * (fs * fss + ft * fst), gt = 2 * (fs * fst + ft * ftt);
    const Decimal H = h * (1 + s / 10 + t / 20), z = (depth - Decimal("0.5")) * H;
    const Decimal zs = (depth - Decimal("0.5")) * h / 10;
    const Decimal zt = (depth - Decimal("0.5")) * h / 20;
    // Antiderivatives of sigma^m k(sigma), independent of the dictionary expansion.
    std::array<Decimal, 4> moments{};
    Decimal power = s;
    for (std::size_t m = 0; m < moments.size(); ++m) {
        const Decimal d = m + 1;
        moments[m] = power / d;
        if (mode == 1)
            moments[m] = 2 * s * power / (d + 1) - moments[m];
        power *= s;
    }
    const auto& [I0, I1, I2, I3] = moments;
    const Decimal k = mode == 0 ? Decimal(1) : Decimal(2 * s - 1);
    const Decimal U = (t / 5 - 3 * t * t / 50) * I0 + t * I2 / 20;  // integral f_t k
    const Decimal V = (Decimal(1) / 5 - 3 * t / 25) * I0 + I2 / 20; // integral f_tt k
    const Decimal W = (Decimal(1) / 5 - 3 * t / 25) * I1 + I3 / 20; // integral sigma f_tt k
    const Decimal b = s * I0 - I1;
    const Decimal A = ft * I0 - U, B = s * V - W;
    const Decimal tx = (1 + fs * fs) * I0 + ft * U;
    const Decimal ty = fs * A + ft * B;
    const Decimal txs = (1 + fs * fs) * k + 2 * fs * fss * I0 + fst * U + ft * ft * k;
    const Decimal txt = 2 * fs * fst * I0 + ftt * U + ft * V;
    const Decimal tys = fss * A + fs * fst * I0 + fst * B + ft * V;
    const Decimal tyt = fst * A + fs * (ftt * I0 - V) + ftt * B - 3 * ft * b / 25;
    const Decimal numerator = -fs * tx - ft * ty;
    const Decimal p = numerator / g32;
    const Decimal ps = (-fss * tx - fs * txs - fst * ty - ft * tys) / g32 - 3 * p * gs / (2 * g);
    const Decimal pt = (-fst * tx - fs * txt - ftt * ty - ft * tyt) / g32 - 3 * p * gt / (2 * g);
    const auto along_s = cross(reference.along_t, reference.along_depth);
    const auto along_t = cross(reference.along_depth, reference.along_s);
    const auto along_depth = cross(reference.along_s, reference.along_t);
    FieldJet output{b + z * p, {}};
    for (std::size_t d = 0; d < output.gradient.size(); ++d) {
        output.gradient[d] = (along_s[d] * (I0 + zs * p + z * ps) + along_t[d] * (zt * p + z * pt) +
                              along_depth[d] * H * p) /
                             reference.volume_density;
    }
    return output;
}

struct WorkPoint {
    std::array<std::vector<PgCartesianExactPosition>, 2> force;
    Decimal expected;
    Decimal root;
};

using RequiredFields = std::array<core::PlannedBuffer<ExactRational>, 2>;
using FreeFields = std::array<std::vector<PgCartesianExactPosition>, 2>;

FreeFields check_supported_tangent(const cc::CasePgShellReference& shell,
                                   const RequiredFields& modes, const PgReferenceLimits& limits,
                                   core::Workspace& workspace, core::ResourcePlan& plan) {
    const auto& chart = shell.chart();
    const auto& original = chart.original();
    const auto& identity = shell.chart_identity();
    const auto free_columns = chart.free_columns();
    Fixture fixture;
    FreeFields free;
    for (std::size_t k = 0; k < modes.size(); ++k) {
        const auto coefficients = modes[k].values();
        std::vector<PgCartesianExactPosition> full(coefficients.size());
        for (std::size_t j = 0; j < coefficients.size(); ++j)
            full[j][0] = owned(fixture, q(coefficients[j]));
        free[k].resize(free_columns.size());
        for (std::size_t j = 0; j < free_columns.size(); ++j)
            free[k][j][0] =
                owned(fixture, q(coefficients[static_cast<std::size_t>(free_columns[j])]));
        const auto original_value =
            take(original.apply({original.layout(), full}, limits, workspace, plan));
        for (const auto& row : original_value.values())
            for (const auto& value : row)
                CHECK(value.sign() == 0);
        const auto tangent = take(chart.tangent({identity, free[k]}, limits, workspace, plan));
        const auto x0 = shell.patch().reference_coordinates();
        for (std::size_t j = 0; j < free_columns.size(); ++j)
            for (std::size_t d = 0; d < x0[j].size(); ++d)
                free[k][j][d] = owned(fixture, q(x0[static_cast<std::size_t>(free_columns[j])][d]) +
                                                   q(free[k][j][d]));
        const auto lift = take(chart.lift({identity, free[k]}, limits, workspace, plan));
        REQUIRE(tangent.size() == coefficients.size());
        REQUIRE(lift.size() == coefficients.size());
        for (std::size_t j = 0; j < coefficients.size(); ++j) {
            for (std::size_t d = 0; d < x0[j].size(); ++d) {
                CHECK(tangent.values()[j][d] == full[j][d]);
                CHECK(q(lift.values()[j][d]) - q(x0[j][d]) == q(full[j][d]));
            }
        }
        for (std::size_t j = 0; j < free_columns.size(); ++j)
            for (std::size_t d = 0; d < x0[j].size(); ++d)
                free[k][j][d] =
                    owned(fixture,
                          q(free[k][j][d]) - q(x0[static_cast<std::size_t>(free_columns[j])][d]));
    }
    return free;
}

struct Observed {
    Decimal position_error{}, gradient_error{}, rounded_position_error{}, rounded_gradient_error{};
    std::uint64_t count{};
};

Decimal field_value(const Quadratic& value, const Decimal& root) {
    return decimal(value[0]) + decimal(value[1]) * root;
}

const std::array<Q, 4> dual_probe{Q{2}, Q{3, 7}, Q{-4, 7}, Q{6, 7}};

std::optional<WorkPoint> compare_point(const cc::CasePgShellReference& shell,
                                       const RequiredFields& modes, const PgReferencePoint& point,
                                       const Point& xy, const Decimal& depth, const Decimal& h,
                                       const PgReferenceLimits& limits, core::ResourcePlan& plan,
                                       Observed& observed, bool capture_work = false) {
    const auto result = take(shell.patch().evaluate_reference(std::span{&point, 1}, limits, plan));
    const auto cols = result.columns.values();
    REQUIRE(result.points.size() == 1);
    REQUIRE(result.points.values().front().layer == core::LayerId{1});
    const Decimal root = sqrt(decimal(result.points.values().front().g));
    std::array<std::array<Quadratic, 4>, 2> actual{};
    std::array<std::array<Q, 4>, 2> rounded{};
    std::optional<WorkPoint> work_point;
    if (capture_work) {
        work_point.emplace();
        work_point->root = root;
        for (auto& part : work_point->force)
            part.resize(cols.size());
    }
    Fixture fixture;
    for (std::size_t j = 0; j < cols.size(); ++j) {
        const std::array<Q, 2> coefficients{q(modes[0].values()[j]), q(modes[1].values()[j])};
        if (!work_point && coefficients[0] == 0 && coefficients[1] == 0)
            continue;
        const auto& column = cols[j];
        const std::array<const PgReferenceScalar*, 4> shape{
            &column.value, &column.physical_gradient_per_metre[0],
            &column.physical_gradient_per_metre[1], &column.physical_gradient_per_metre[2]};
        Quadratic force{};
        for (std::size_t a = 0; a < shape.size(); ++a) {
            const auto& value = *shape[a];
            const Quadratic qvalue{q(value.rational), q(value.radical)};
            const Decimal v = field_value(qvalue, root);
            const double hi = v.convert_to<double>();
            const double lo = (v - decimal(Q{hi})).convert_to<double>();
            const Q two_word = Q{hi} + Q{lo};
            for (std::size_t field = 0; field < modes.size(); ++field) {
                const auto& coeff = coefficients[field];
                actual[field][a][0] += qvalue[0] * coeff;
                actual[field][a][1] += qvalue[1] * coeff;
                rounded[field][a] += two_word * coeff;
            }
            if (work_point) {
                force[0] += dual_probe[a] * qvalue[0];
                force[1] += dual_probe[a] * qvalue[1];
            }
        }
        if (work_point)
            for (std::size_t part = 0; part < force.size(); ++part)
                work_point->force[part][j][0] = owned(fixture, force[part]);
    }
    const Decimal independent_tolerance("1e-65");
    const Decimal reproduction_tolerance("1e-8"); // Adopted §5, L_ref=1 m and 1 m amplitude.
    for (std::size_t k = 0; k < modes.size(); ++k) {
        const auto expected = independent_field(xy, depth, h, k);
        const std::array<Decimal, 4> oracle{expected.value, expected.gradient[0],
                                            expected.gradient[1], expected.gradient[2]};
        for (std::size_t a = 0; a < oracle.size(); ++a) {
            const Decimal error = abs(field_value(actual[k][a], root) - oracle[a]);
            const Decimal projected_error = abs(decimal(rounded[k][a]) - oracle[a]);
            CHECK(error <= independent_tolerance);
            CHECK(projected_error <= reproduction_tolerance);
            auto& exact_report = a == 0 ? observed.position_error : observed.gradient_error;
            auto& rounded_report =
                a == 0 ? observed.rounded_position_error : observed.rounded_gradient_error;
            exact_report = (std::max)(exact_report, error);
            rounded_report = (std::max)(rounded_report, projected_error);
        }
        if (work_point && k == 0)
            for (std::size_t a = 0; a < oracle.size(); ++a)
                work_point->expected += decimal(dual_probe[a]) * oracle[a];
    }
    ++observed.count;
    return work_point;
}

void check_virtual_work(const cc::CasePgShellReference& shell, const RequiredFields& fields,
                        const FreeFields& free, const WorkPoint& point,
                        const PgReferenceLimits& limits, core::Workspace& workspace,
                        core::ResourcePlan& plan) {
    const auto& chart = shell.chart();
    Fixture fixture;
    std::vector<PgCartesianExactPosition> full(fields.front().size());
    for (std::size_t j = 0; j < full.size(); ++j)
        full[j][0] = owned(fixture, q(fields.front().values()[j]));
    Quadratic observed{};
    for (std::size_t part = 0; part < observed.size(); ++part) {
        const auto& force = point.force[part];
        const auto dual =
            take(chart.transpose({shell.chart_identity(), force}, limits, workspace, plan));
        observed[part] = work(free.front(), dual.independent_force.values());
        CHECK(observed[part] == work(full, force));
    }
    CHECK(abs(point.expected) > Decimal("1e-6"));
    const Decimal error = abs(field_value(observed, point.root) - point.expected);
    CHECK(error <= Decimal("1e-65"));
    std::clog << "R5_PHYSICAL_WORK," << point.expected << ',' << error << '\n' << std::flush;
}

void observe_layout(const cc::CasePgChart& ready, const Decimal& h, core::Workspace& workspace,
                    core::ResourcePlan& plan) {
    const auto shell = take(ready.shell_reference(numerics::PatchId{1}, 1));
    const auto limits = chart_limits(2).chart.arithmetic;
    const auto& condition = shell.patch().conditioning();
    // Required columns 3+2*8+k are the Z component of the global s curvature.
    const RequiredFields fields{
        take(condition.required_coordinates({19}, limits, workspace, plan)),
        take(condition.required_coordinates({20}, limits, workspace, plan))};
    const auto free = check_supported_tangent(shell, fields, limits, workspace, plan);
    const auto& partition = take(take(condition.reference_basis())->source()).partition;
    const auto triangles = take(partition.triangles());
    const auto vertices = take(partition.vertices());
    const auto volumes = take(partition.volumes());
    const auto horizontal = take(partition.horizontal_faces());
    const auto lateral = take(partition.lateral_faces());
    const auto edges = take(partition.edges());
    Fixture fixture;
    const auto third = fixture.r("1", "3"), half = fixture.r("1", "2");
    const auto shallow = fixture.r("1", "4"), deep = fixture.r("3", "4");
    const Decimal third_decimal = Decimal(1) / 3;
    Observed observations;
    for (std::size_t i = 0; i < volumes.size(); ++i) {
        const auto& record = volumes[i];
        const auto& triangle = triangles[static_cast<std::size_t>(record.triangle)];
        const auto xy = barycentric(vertices, triangle, third_decimal, third_decimal);
        const auto& depth = i % 2 == 0 ? shallow : deep;
        const PgReferencePoint query{
            PgReferenceVolumePoint{static_cast<core::Index>(i), third, third, depth}};
        const auto dual = compare_point(shell, fields, query, xy, decimal(depth), h, limits, plan,
                                        observations, i == volumes.size() / 2);
        if (dual)
            check_virtual_work(shell, fields, free, *dual, limits, workspace, plan);
    }
    for (std::size_t i = 0; i < horizontal.size(); ++i) {
        const auto& record = horizontal[i];
        const auto& triangle = triangles[static_cast<std::size_t>(record.triangle)];
        const auto xy = barycentric(vertices, triangle, third_decimal, third_decimal);
        const Decimal depth = record.side == PgPartitionFaceSide::Lower ? 0 : 1;
        const PgReferencePoint query{
            PgReferenceHorizontalPoint{static_cast<core::Index>(i), third, third}};
        static_cast<void>(
            compare_point(shell, fields, query, xy, depth, h, limits, plan, observations));
    }
    for (std::size_t i = 0; i < lateral.size(); ++i) {
        const auto& edge = edges[static_cast<std::size_t>(lateral[i].edge)];
        const auto a = point(vertices[static_cast<std::size_t>(edge.vertices[0])]);
        const auto b = point(vertices[static_cast<std::size_t>(edge.vertices[1])]);
        const Point xy{(a[0] + b[0]) / 2, (a[1] + b[1]) / 2};
        const auto& depth = i % 2 == 0 ? shallow : deep;
        const PgReferencePoint query{
            PgReferenceLateralPoint{static_cast<core::Index>(i), half, depth}};
        static_cast<void>(
            compare_point(shell, fields, query, xy, decimal(depth), h, limits, plan, observations));
    }
    CHECK(observations.count == volumes.size() + horizontal.size() + lateral.size());
    std::clog << "R5_PHYSICAL_OBSERVATION," << volumes.size() << ',' << horizontal.size() << ','
              << lateral.size() << ',' << observations.position_error << ','
              << observations.gradient_error << ',' << observations.rounded_position_error << ','
              << observations.rounded_gradient_error << ',' << plan.allocated_bytes() << '\n'
              << std::flush;
}

TEST_CASE("R5 supported nonaffine fields preserve physical gradients and virtual work",
          "[.r5-physical-reference][V-PG][SH-D03][SH-D05][SH-D07][verification]") {
    for (core::Index i = 0; i < 4; ++i) {
        DYNAMIC_SECTION("R5 " << i) {
            const auto path = std::string("r5-taper-0") + char('0' + i) + "/case.json";
            std::ifstream stream(path, std::ios::binary);
            REQUIRE(stream);
            const std::string wire{std::istreambuf_iterator<char>{stream},
                                   std::istreambuf_iterator<char>{}};
            const auto n = i < 2 ? 1 : 2;
            const auto h = i % 2 == 0 ? Decimal("0.01") : Decimal("0.0001");
            core::Workspace workspace{core::PlannedBuffer<std::byte>{}};
            core::ResourcePlan plan{profile};
            const auto ready = prepare_r5(wire, n, workspace, plan);
            std::clog << "R5_PHYSICAL_BEGIN," << i << ',' << plan.allocated_bytes() << '\n'
                      << std::flush;
            observe_layout(ready, h, workspace, plan);
        }
    }
}
