def derive(original):
    if hashlib.sha256(original).hexdigest() != BASE_SHA256:
        raise ValueError('frozen C++ source SHA differs')
    result = original
    for before, after in EDITS:
        old, new = before.encode(), after.encode()
        if result.count(old) != 1 or new in result:
            raise ValueError(f'derivation anchor differs: {before}')
        result = result.replace(old, new)
    restored = result
    for before, after in reversed(EDITS):
        restored = restored.replace(after.encode(), before.encode())
    if restored != original:
        raise ValueError('reverse derivation differs from the frozen source')
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('base', 'guard', 'output', 'record'):
        parser.add_argument(f'--{name}', type=Path, required=True)
    args = parser.parse_args()
    run_path(str(args.guard))['distinct_paths'](args.base, args.guard, args.output, args.record)
    source = derive(args.base.read_bytes())
    args.output.write_bytes(source)
    args.record.write_text(json.dumps({
        'schema': 'uvlas.pg-shared-fixed-batch-derivation/1',
        'base_sha256': BASE_SHA256,
        'generated_header_sha256': hashlib.sha256(source).hexdigest(),
        'generated_bytes': len(source),
        'edits': len(EDITS),
    }, indent=2) + '\n', encoding='utf-8', newline='\n')


if __name__ == '__main__':
    main()
