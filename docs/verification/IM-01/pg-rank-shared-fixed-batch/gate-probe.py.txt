"""Test combined publication using captured aggregates and no numerical subprocess."""

import argparse
from copy import deepcopy
import json
from pathlib import Path
from runpy import run_path
from shutil import copyfile
import subprocess
import sys
from tempfile import TemporaryDirectory
from unittest.mock import patch


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('root', 'coefficient', 'full-certificate', 'cut-certificate', 'capture'):
        parser.add_argument(f'--{name}', type=Path, required=True)
    args = parser.parse_args()
    verification = args.root.resolve() / 'docs/verification/IM-01'
    folder = verification / 'pg-rank-shared-fixed-batch'
    run = run_path(str(folder / 'run-shared.py.txt'))['main']
    capture = json.loads(args.capture.read_bytes())
    captures = {case['domain']: case for case in capture['cases']}
    ordinary_calls = ('--controls', '--shared-controls', 'full:first', 'full:all', 'cut:first', 'cut:all')
    scenarios = (
        ('valid', None, ordinary_calls),
        ('late-cut-failure', 'injected cut failure', ordinary_calls),
        ('cut-digest-mismatch', 'frozen numerical field differs: column_diagnostics_sha256', ordinary_calls),
        ('swapped-certificate', 'full input identity', ()),
        ('swapped-baseline', 'frozen identity:', ()),
    )
    paths = {
        'coefficient': args.coefficient, 'full-certificate': args.full_certificate,
        'cut-certificate': args.cut_certificate,
        'certificate-index': verification / 'pg-rank-fixed-certificate/certificate-index.json',
        'study': verification / 'pg-rank-resource-study/results.json',
        'derivation': folder / 'derivation.json',
        'baseline-full': verification / 'pg-rank-fixed-batch-n4-full/quartic-result.json',
        'baseline-cut': verification / 'pg-rank-fixed-batch-n4-cut/results.json',
        'shared-census': verification / 'pg-rank-shared-pivot-scale/result.json',
        'base-runner': verification / 'pg-rank-fixed-batch/run-batch.py.txt',
        'cut-runner': verification / 'pg-rank-fixed-batch-n4-cut/run-case.py.txt',
        'full-runner': verification / 'pg-rank-fixed-batch-n4-full/run-case.py.txt',
        'capacity-producer': verification / 'pg-rank-fixed-slot-design/capacity-oracle-v2.py.txt',
    }
    cut = run_path(str(paths['cut-runner']))
    full = run_path(str(paths['full-runner']))
    load = cut['load']
    base = load('gate_base', paths['base-runner'])
    deriver = folder / 'derive-header.py.txt'
    original_source = (verification / 'pg-rank-fixed-batch/fixed-batch.cpp.txt').read_bytes()
    derive = run_path(str(deriver))['derive']
    derived = derive(original_source)
    base.integer(len(derived), capture['derivation']['generated_bytes'], 'generated length')
    try:
        derive(original_source + b' ')
    except ValueError:
        pass
    else:
        raise RuntimeError('accepted modified frozen source')

    with TemporaryDirectory() as scratch:
        temp = Path(scratch)
        copies = {name: temp / (name + path.suffix) for name, path in paths.items()}
        for name, path in paths.items():
            copyfile(path, copies[name])
        copies['binary'] = temp / 'batch.exe'
        copies['binary'].write_bytes(b'publication test transport')
        output = temp / 'result.json'
        old = b'prior complete result\n'
        calls = []

        def argv(inputs, destination):
            return [atom for key, path in inputs.items() for atom in (f'--{key}', str(path))] + [
                '--output', str(destination),
            ]

        def load_for_gate(name, path):
            return base if name == 'shared_batch_base' else load(name, path)

        def run_path_for_gate(path):
            # The runtime capture has already been checked per column and compacted.
            if Path(path) == copies['cut-runner']:
                return {**cut, 'load': load_for_gate, 'compact': lambda result: result}
            return {**full, 'validate_batch': lambda *args: None}

        with patch.dict(run.__globals__, run_path=run_path_for_gate):
            for present in (False, True):
                for name, expected, order in scenarios:
                    inputs = dict(copies)
                    if name == 'swapped-certificate':
                        inputs['full-certificate'] = copies['cut-certificate']
                    if name == 'swapped-baseline':
                        inputs['baseline-full'] = copies['baseline-cut']
                    output.unlink(missing_ok=True)
                    if present:
                        output.write_bytes(old)
                    calls.clear()

                    def invoke(_binary, *values):
                        if len(values) == 1:
                            calls.append(values[0])
                            return capture['shared_controls' if values[0] == '--shared-controls' else 'controls']
                        domain, mode = values[-2:]
                        calls.append(f'{domain}:{mode}')
                        if name == 'late-cut-failure' and (domain, mode) == ('cut', 'all'):
                            raise RuntimeError('injected cut failure')
                        result = deepcopy(captures[domain]['first_probe' if mode == 'first' else 'result'])
                        if name == 'cut-digest-mismatch' and (domain, mode) == ('cut', 'all'):
                            result['column_diagnostics_sha256'] = '0' * 64
                        return result

                    with patch.object(base, 'invoke', side_effect=invoke):
                        try:
                            run(argv(inputs, output))
                        except RuntimeError as error:
                            base.require(expected is not None and str(error).startswith(expected), f'{name}: {error}')
                            base.require(output.read_bytes() == old if present else not output.exists(),
                                         f'{name}: output changed')
                        else:
                            base.require(expected is None, f'{name}: incorrectly published')
                            published = json.loads(output.read_bytes())
                            base.require(published['schema'] == capture['schema']
                                         and published['cases'] == capture['cases'], 'complete published case pair')
                    base.require(tuple(calls) == order and not list(temp.glob('.*.tmp')), f'{name}: order or temp')

            alias = temp / 'output-alias'
            with patch.object(base, 'invoke', side_effect=AssertionError('probe invoked for alias')):
                for name, target in copies.items():
                    before = target.read_bytes()
                    alias.hardlink_to(target)
                    for collision in (target, alias):
                        try:
                            run(argv(copies, collision))
                        except ValueError as error:
                            base.require(str(error) == f'--output must be distinct from read-only input: {target}',
                                         f'{name}: {error}')
                        else:
                            raise RuntimeError(f'{name}: accepted aliased output')
                        base.require(alias.samefile(target) and target.read_bytes() == before
                                     and not list(temp.glob('.*.tmp')), f'{name}: input changed')
                    alias.unlink()

        original = temp / 'frozen.cpp'
        original.write_bytes(original_source)
        generated = temp / 'generated.hpp'
        generated.hardlink_to(original)
        guard = verification / 'pg-rank-fixed-batch-n2/derive-driver.py.txt'
        record = temp / 'derive.json'
        optimize = ['-O'] if sys.flags.optimize else []
        result = subprocess.run([
            sys.executable, *optimize, '-B', deriver, '--base', original, '--guard', guard,
            '--output', generated, '--record', record,
        ], capture_output=True, timeout=15)
        base.require(result.returncode != 0 and b'distinct files' in result.stderr
                     and generated.samefile(original) and original.read_bytes() == original_source
                     and not record.exists(), 'derivation input collision')

    print(json.dumps({
        'schema': 'uvlas.pg-shared-fixed-batch-gate/1',
        'publication_checks': len(scenarios) * 2,
        'publication_scenarios': [name for name, _, _ in scenarios],
        'runner_alias_checks': len(copies) * 2, 'derivation_alias_checks': 1,
        'modified_derivation_source_rejected': True,
    }, indent=2))


if __name__ == '__main__':
    main()
