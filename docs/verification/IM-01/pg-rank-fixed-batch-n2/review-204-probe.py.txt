"""Reproduce review #204 and test only the two changed entrypoint boundaries."""
import argparse
import copy
import hashlib
import importlib.machinery
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile


def require(value,message):
    if not value:raise RuntimeError(message)


def load(name,path):
    loader=importlib.machinery.SourceFileLoader(name,str(path));spec=importlib.util.spec_from_loader(name,loader)
    module=importlib.util.module_from_spec(spec);loader.exec_module(module);return module


def main():
    p=argparse.ArgumentParser();p.add_argument('--root',type=Path,required=True);p.add_argument('--output',type=Path,required=True)
    a=p.parse_args();root=a.root.resolve();folder=root/'docs/verification/IM-01/pg-rank-fixed-batch-n2'
    source=(root/'docs/verification/IM-01/pg-rank-fixed-batch/fixed-batch.cpp.txt').read_bytes()
    old=folder/'review-204-original-derive-driver.py.txt';new=folder/'derive-driver.py.txt'
    flags=['-O'] if sys.flags.optimize else []
    path_results=[];old_results=[]
    with tempfile.TemporaryDirectory() as temp:
        d=Path(temp)
        for script,is_old in [(old,True),(new,False)]:
            for pair in [('base','output'),('base','record'),('output','record')]:
                for alias in (['direct'] if is_old else ['direct','relative','hardlink']):
                    paths={'base':d/'base.cpp','output':d/'generated.cpp','record':d/'driver.json'}
                    for name in ['base.cpp','generated.cpp','driver.json','alias']:
                        (d/name).unlink(missing_ok=True)
                    paths['base'].write_bytes(source);paths['output'].write_bytes(b'old generated');paths['record'].write_bytes(b'old record')
                    target=paths[pair[0]]
                    if alias=='direct':paths[pair[1]]=target
                    elif alias=='relative':
                        (d/'child').mkdir(exist_ok=True);paths[pair[1]]=d/'child'/'..'/target.name
                    else:
                        os.link(target,d/'alias');paths[pair[1]]=d/'alias'
                    before={path:path.read_bytes() for path in paths.values()}
                    command=[sys.executable,*flags,'-B',str(script),'--base',str(paths['base']),
                        '--output',str(paths['output']),'--record',str(paths['record'])]
                    run=subprocess.run(command,capture_output=True,timeout=10)
                    if is_old:
                        require(run.returncode==0 and any(path.read_bytes()!=data for path,data in before.items()),'legacy collision reproduction')
                        old_results.append({'pair':list(pair),'accepted':True,'existing_file_changed':True})
                    else:
                        require(run.returncode!=0 and all(path.read_bytes()==data for path,data in before.items()),'collision refused before all writes')
                        path_results.append({'pair':list(pair),'alias':alias,'all_bytes_preserved':True})
        # Distinct paths still reproduce exactly the frozen generated bytes and metadata.
        control=d/'normal';control.mkdir()
        base_path=control/'base.cpp';base_path.write_bytes(source)
        run=subprocess.run([sys.executable,*flags,'-B',str(new),'--base',str(base_path),'--output',str(control/'derived.cpp'),
                            '--record',str(control/'driver.json')],capture_output=True,timeout=10)
        require(run.returncode==0 and base_path.read_bytes()==source,'distinct paths normal')
        require(hashlib.sha256((control/'derived.cpp').read_bytes()).hexdigest()==json.loads((folder/'driver.json').read_text())['generated_source_sha256'],'normal derived bytes')
        base=load('shared_review_base',root/'docs/verification/IM-01/pg-rank-fixed-batch/run-batch.py.txt')
        case=load('new_review_case',folder/'run-case.py.txt');legacy=load('old_review_case',folder/'review-204-original-run-case.py.txt')
        capture=json.loads((folder/'results.json').read_text());raw=json.loads((root/'out/issue140-n2-full/n2-full-raw.json').read_text())
        first={**capture['first_probe'],'columns':[{'proof_rows':0} for _ in range(730)]}
        chosen=set(capture['result']['selected_columns']);j=next(k for k in range(730) if k not in chosen)
        first['columns'][j]=copy.deepcopy(raw['columns'][j])
        valid=json.loads((folder/'driver.json').read_text());variants=[('valid',valid),('empty',{}),('array',[]),('null',None),('invalid-json','INVALID JSON')]
        for key,expected in valid.items():
            missing=copy.deepcopy(valid);missing.pop(key);variants.append((key+'-missing',missing))
            mismatch=copy.deepcopy(valid);mismatch[key]=False if type(expected) is bool else (expected+1 if type(expected) is int else 'incorrect')
            variants.append((key+'-mismatch',mismatch))
            wrong=copy.deepcopy(valid);wrong[key]=1 if type(expected) is bool else (float(expected) if type(expected) is int else None)
            variants.append((key+'-type',wrong))
        exe=d/'batch.exe';exe.write_bytes(b'numeric transport is stubbed');output=d/'result.json';record=d/'request-driver.json'
        def argv():
            return ['--inputs',str(root/'out/issue140-scratch'),'--certificates',str(root/'out/issue164-fixed-verifier'),
                '--batch',str(exe),'--base-runner',str(root/'docs/verification/IM-01/pg-rank-fixed-batch/run-batch.py.txt'),
                '--capacity-producer',str(root/'docs/verification/IM-01/pg-rank-fixed-slot-design/capacity-oracle-v2.py.txt'),
                '--study',str(root/'docs/verification/IM-01/pg-rank-resource-study/results.json'),
                '--certificate-index',str(root/'docs/verification/IM-01/pg-rank-fixed-certificate/certificate-index.json'),
                '--driver-record',str(record),'--output',str(output)]
        def wire(module):
            original_load=module.load
            module.load=lambda name,path:base if name=='batch_base' else original_load(name,path)
        wire(case);wire(legacy)
        calls=[]
        def invoke(binary,*args):
            calls.append(args[-1]);return capture['controls'] if args==('--controls',) else (copy.deepcopy(first) if args[-1]=='first' else copy.deepcopy(raw))
        base.invoke=invoke
        record.write_text('{}');legacy.main(argv())
        require(json.loads(output.read_text())['driver']=={} and len(calls)==3,'legacy empty record accepted after all transport')
        driver_results=[]
        for present in [False,True]:
            for name,value in variants:
                output.unlink(missing_ok=True)
                if present:output.write_bytes(b'previous completed result\n')
                record.write_text(value if name=='invalid-json' else json.dumps(value))
                calls.clear();accepted=False
                try:case.main(argv());accepted=True
                except (RuntimeError,ValueError):pass
                require(accepted==(name=='valid'),name+' outcome')
                if accepted:
                    require(json.loads(output.read_text())['driver']==valid and len(calls)==3,'validated record normal')
                else:
                    require(not calls,'invalid provenance before all numeric transport')
                    require(output.read_bytes()==b'previous completed result\n' if present else not output.exists(),'previous result preserved')
                require(not list(d.glob('.result.json.*.tmp')),'no temporary results')
                driver_results.append({'scenario':name,'previous_output':present,'accepted':accepted,'numeric_transport_calls':len(calls)})
    a.output.write_text(json.dumps({'schema':'uvlas.review-204/1','numeric_transport':'captured-result stub',
        'legacy_collisions':old_results,'legacy_empty_record_published':True,'distinct_paths_reproduce':True,
        'path_collision_controls':path_results,'driver_controls':driver_results},indent=2)+'\n',encoding='utf-8',newline='\n')


if __name__=='__main__':main()
