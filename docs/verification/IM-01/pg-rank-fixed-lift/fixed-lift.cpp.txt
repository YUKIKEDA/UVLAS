#include "allocation-profile.hpp.txt"

namespace lift_fixed {
using Small = bounded::Int<512>;
using Big = bounded::Int<8192>;
constexpr std::uint64_t prime_value = p1;
struct Pair {
    Big n{0}, d{1};
};
struct IntegerTerm {
    std::uint32_t col;
    Small value;
};
struct ModularTerm {
    std::uint32_t index, value;
};
template <class T>
std::uint64_t bit_length(const T& v) {
    if (v == 0)
        return 0;
    T positive = v;
    if (positive < 0)
        positive = -positive;
    return 1 + boost::multiprecision::msb(positive);
}
std::uint64_t plus(std::uint64_t a, std::uint64_t b) {
    if (b > UINT64_MAX - a)
        throw std::overflow_error("payload sum");
    return a + b;
}
std::uint64_t times(std::uint64_t a, std::uint64_t b) {
    if (b && a > UINT64_MAX / b)
        throw std::overflow_error("payload product");
    return a * b;
}
struct Data {
    std::uint32_t candidate{};
    std::vector<std::uint32_t> selected, scale_inverse, diagonal_inverse;
    std::vector<std::uint64_t> row_offsets, lower_offsets, upper_offsets;
    std::vector<IntegerTerm> matrix;
    std::vector<ModularTerm> lower, upper;
    std::vector<Small> initial, bound;
    std::vector<Pair> expected;
    std::uint64_t matrix_bits{}, row_bound_bits{}, product_bound_bits{};
};
Data prepare(const Matrix& m, const Elimination<Mod<p1>>& factor, std::uint32_t candidate) {
    if (!factor.complete || candidate >= m.columns.size() || !factor.coordinates[candidate])
        throw std::runtime_error("incomplete factor or nondependent target");
    const auto r = factor.selected.size();
    Data d;
    d.candidate = candidate;
    d.selected.assign(factor.selected.begin(), factor.selected.end());
    d.initial.resize(r);
    d.bound.resize(r);
    d.scale_inverse.resize(r);
    d.diagonal_inverse.resize(r);
    d.expected.resize(r);
    d.row_offsets.push_back(0);
    d.lower_offsets.push_back(0);
    d.upper_offsets.push_back(0);
    std::vector<int> mapping(m.rows, -1);
    for (std::size_t k = 0; k < r; ++k)
        mapping[factor.pivot_rows[k]] = static_cast<int>(k);
    std::vector<Sparse<Q>> rows(r);
    for (std::size_t k = 0; k < r; ++k)
        for (const auto& [row, value] : m.columns[factor.selected[k]])
            if (mapping[row] >= 0)
                rows[mapping[row]].emplace_back(static_cast<int>(k), value);
    for (const auto& [row, value] : m.columns[candidate])
        if (mapping[row] >= 0)
            rows[mapping[row]].emplace_back(-1, value);
    for (std::size_t row = 0; row < r; ++row) {
        Integer scale = 1;
        for (const auto& [col, q] : rows[row])
            scale = (scale / gcd(scale, Integer{denominator(q)})) * denominator(q);
        if (scale % p1 == 0)
            throw std::runtime_error("noninvertible row scale");
        d.scale_inverse[row] =
            static_cast<std::uint32_t>((Mod<p1>{1} / Mod<p1>::parse(Q{scale})).value);
        Integer sum = 0, rhs = 0;
        for (const auto& [col, q] : rows[row]) {
            const Integer value = numerator(q) * (scale / denominator(q));
            if (col < 0) {
                rhs = value;
                continue;
            }
            if (value == 0)
                continue;
            if (bit_length(value) > 512)
                throw std::runtime_error("matrix bit admission");
            d.matrix.push_back({static_cast<std::uint32_t>(col), Small{value}});
            sum += boost::multiprecision::abs(value);
            d.matrix_bits = std::max(d.matrix_bits, bit_length(value));
        }
        const Integer magnitude = boost::multiprecision::abs(rhs);
        const Integer bound = std::max(sum, magnitude);
        const auto bound_bits = bit_length(bound);
        if (bound_bits + 32 > 512)
            throw std::runtime_error("current recurrence bit admission");
        d.initial[row] = Small{rhs};
        d.bound[row] = Small{bound};
        d.row_bound_bits = std::max(d.row_bound_bits, bound_bits);
        d.product_bound_bits = std::max(d.product_bound_bits, bound_bits + 32);
        d.row_offsets.push_back(d.matrix.size());
        for (const auto& [index, value] : factor.pivot_lower[row]) {
            if (index <= static_cast<int>(row))
                throw std::runtime_error("lower factor shape");
            d.lower.push_back(
                {static_cast<std::uint32_t>(index), static_cast<std::uint32_t>(value.value)});
        }
        d.lower_offsets.push_back(d.lower.size());
        if (factor.pivot_upper[row].empty() ||
            factor.pivot_upper[row].back().first != static_cast<int>(row))
            throw std::runtime_error("upper diagonal shape");
        d.diagonal_inverse[row] =
            static_cast<std::uint32_t>((Mod<p1>{1} / factor.pivot_upper[row].back().second).value);
        for (std::size_t j = 0; j + 1 < factor.pivot_upper[row].size(); ++j) {
            const auto& [index, value] = factor.pivot_upper[row][j];
            if (index >= static_cast<int>(row))
                throw std::runtime_error("upper factor shape");
            d.upper.push_back(
                {static_cast<std::uint32_t>(index), static_cast<std::uint32_t>(value.value)});
        }
        d.upper_offsets.push_back(d.upper.size());
    }
    return d;
}
void expected_coordinates(Data& d, const char* path, int grid, int rows, int full, int columns,
                          const Elimination<Mod<p1>>& factor) {
    std::ifstream stream(path);
    std::string tag, numerator_text, denominator_text;
    int n, m, f, count, r;
    if (!(stream >> tag >> n >> m >> f >> count >> r) || tag != "certificate" || n != grid ||
        m != rows || f != full || count != columns || r != static_cast<int>(d.selected.size()))
        throw std::runtime_error("independent certificate identity");
    for (int k = 0; k < r; ++k) {
        int index, column, pivot;
        if (!(stream >> tag >> index >> column >> pivot) || tag != "basis" || index != k ||
            column != factor.selected[k] || pivot != factor.pivot_rows[k])
            throw std::runtime_error("independent certificate factor order");
    }
    for (int j = 0; j < columns; ++j) {
        int column, terms;
        if (!(stream >> tag >> column >> terms) || tag != "column" || column != j || terms < 0 ||
            terms > r)
            throw std::runtime_error("independent certificate column shape");
        int previous = -1;
        for (int k = 0; k < terms; ++k) {
            int index;
            if (!(stream >> index >> numerator_text >> denominator_text) || index <= previous ||
                index >= r)
                throw std::runtime_error("independent coordinate shape");
            previous = index;
            if (j == static_cast<int>(d.candidate)) {
                d.expected[index] = {Big{numerator_text}, Big{denominator_text}};
                if (d.expected[index].n == 0 || d.expected[index].d <= 0 ||
                    d.selected[index] >= d.candidate)
                    throw std::runtime_error("independent coordinate prefix or sign");
            }
        }
    }
    if (stream >> tag)
        throw std::runtime_error("extra independent certificate record");
}
struct Recovery {
    Big r0{}, r1{}, t0{}, t1{}, quotient{}, next{}, check{};
};
struct Scratch {
    std::vector<Small> current;
    std::vector<Big> residue, scaled;
    std::vector<Pair> recovered;
    std::vector<std::uint32_t> rhs, solution;
    Big modulus{1}, half{}, recovery_bound{}, common{1}, g{}, temporary{}, sum{}, term{}, right{};
    Recovery recovery;
    explicit Scratch(const Data& d)
        : current(d.selected.size()),
          residue(current.size()),
          scaled(current.size()),
          recovered(current.size()),
          rhs(current.size()),
          solution(current.size()) {}
};
std::uint64_t input_payload(const Data& d) {
    std::uint64_t bytes = sizeof(Data);
    const auto charge = [&](auto const& v) {
        bytes =
            plus(bytes, times(v.capacity(),
                              sizeof(typename std::remove_reference_t<decltype(v)>::value_type)));
    };
    charge(d.selected);
    charge(d.scale_inverse);
    charge(d.diagonal_inverse);
    charge(d.row_offsets);
    charge(d.lower_offsets);
    charge(d.upper_offsets);
    charge(d.matrix);
    charge(d.lower);
    charge(d.upper);
    charge(d.initial);
    charge(d.bound);
    charge(d.expected);
    return bytes;
}
std::uint64_t plan(const Data& d) {
    return plus(plus(input_payload(d), sizeof(Scratch)),
                times(d.selected.size(),
                      sizeof(Small) + 2 * sizeof(Big) + sizeof(Pair) + 2 * sizeof(std::uint32_t)));
}
std::uint64_t payload(const Data& d, const Scratch& s) {
    auto bytes = plus(input_payload(d), sizeof(Scratch));
    const auto charge = [&](auto const& v) {
        bytes =
            plus(bytes, times(v.capacity(),
                              sizeof(typename std::remove_reference_t<decltype(v)>::value_type)));
    };
    charge(s.current);
    charge(s.residue);
    charge(s.scaled);
    charge(s.recovered);
    charge(s.rhs);
    charge(s.solution);
    return bytes;
}
enum class Status {
    Success,
    PayloadExceeded,
    BitExceeded,
    EventExceeded,
    DigitsExceeded,
    InvalidFactor,
    NonDivisible,
    FutureBasis,
    ReferenceMismatch,
    HeapAttempt,
    ArithmeticFailure,
    InvariantFailure
};
struct Limits {
    std::uint64_t payload, bits, digits, events;
};
struct Report {
    Status status{Status::DigitsExceeded};
    std::uint64_t digits{}, events{}, products{}, triangular_updates{}, recovery_attempts{},
        negative_denominators{}, maximum_current_bits{}, maximum_predivision_bits{},
        maximum_residue_bits{}, maximum_coordinate_bits{}, heap{};
};
bool event(Report& r, const Limits& cap, std::uint64_t count = 1) {
    if (count > cap.events - r.events) {
        r.status = Status::EventExceeded;
        return false;
    }
    r.events += count;
    return true;
}
std::uint32_t reduce(const Small& value) {
    Small remainder = value % p1;
    if (remainder < 0)
        remainder += p1;
    return static_cast<std::uint32_t>(remainder);
}
bool reconstruct(const Big& residue, const Big& modulus, const Big& bound, Pair& out, Scratch& s,
                 Report& r, const Limits& cap) {
    auto& q = s.recovery;
    q.r0 = modulus;
    q.r1 = residue;
    q.t0 = 0;
    q.t1 = 1;
    while (q.r1 > bound) {
        if (!event(r, cap))
            return false;
        q.quotient = q.r0 / q.r1;
        q.next = q.r0 % q.r1;
        q.r0 = q.r1;
        q.r1 = q.next;
        q.next = q.t0 - q.quotient * q.t1;
        q.t0 = q.t1;
        q.t1 = q.next;
    }
    if (q.t1 == 0 || q.t1 > bound || q.t1 < -bound)
        return false;
    q.check = q.r1 - q.t1 * residue;
    if (q.check % modulus != 0)
        return false;
    if (q.t1 < 0) {
        q.r1 = -q.r1;
        q.t1 = -q.t1;
        ++r.negative_denominators;
    }
    q.next = q.r1;
    if (q.next < 0)
        q.next = -q.next;
    s.g = boost::multiprecision::gcd(q.next, q.t1);
    out.n = q.r1 / s.g;
    out.d = q.t1 / s.g;
    if (out.d <= 0)
        return false;
    r.maximum_coordinate_bits =
        std::max({r.maximum_coordinate_bits, bit_length(out.n), bit_length(out.d)});
    return true;
}
bool square_certificate(const Data& d, Scratch& s, Report& r, const Limits& cap) {
    s.common = 1;
    std::uint64_t numerator_bits = 0;
    for (const auto& value : s.recovered) {
        if (!event(r, cap))
            return false;
        if (value.n == 0)
            continue;
        s.g = boost::multiprecision::gcd(s.common, value.d);
        s.temporary = s.common / s.g;
        if (bit_length(s.temporary) + bit_length(value.d) > 8192) {
            r.status = Status::BitExceeded;
            return false;
        }
        s.common = s.temporary * value.d;
        numerator_bits = std::max(numerator_bits, bit_length(value.n));
    }
    const auto required = std::max(d.matrix_bits + numerator_bits + bit_length(s.common) +
                                       std::bit_width(d.selected.size()) + 2,
                                   d.row_bound_bits + bit_length(s.common) + 1);
    if (required > 8192) {
        r.status = Status::BitExceeded;
        return false;
    }
    for (std::size_t k = 0; k < s.recovered.size(); ++k) {
        if (!event(r, cap))
            return false;
        if (s.recovered[k].n == 0)
            s.scaled[k] = 0;
        else {
            s.scaled[k] = s.common / s.recovered[k].d;
            s.scaled[k] *= s.recovered[k].n;
        }
    }
    for (std::size_t row = 0; row < s.current.size(); ++row) {
        s.sum = 0;
        for (auto pos = d.row_offsets[row]; pos < d.row_offsets[row + 1]; ++pos) {
            if (!event(r, cap, 2))
                return false;
            const auto& entry = d.matrix[pos];
            s.term = s.scaled[entry.col] * entry.value;
            s.sum += s.term;
        }
        s.right = s.common * d.initial[row];
        if (s.sum != s.right)
            return false;
    }
    return true;
}
Report run(const Data& d, Scratch& s, const Limits& cap) {
    Report r;
    if (payload(d, s) > cap.payload) {
        r.status = Status::PayloadExceeded;
        return r;
    }
    if (cap.bits < 8192 || cap.digits > 128 || d.product_bound_bits > 512) {
        r.status = Status::BitExceeded;
        return r;
    }
    for (auto inverse : d.diagonal_inverse)
        if (!inverse) {
            r.status = Status::InvalidFactor;
            return r;
        }
    for (auto inverse : d.scale_inverse)
        if (!inverse) {
            r.status = Status::InvalidFactor;
            return r;
        }
    bounded::heap_attempts = 0;
    bounded::watched = true;
    try {
        const auto body = [&]() {
            std::copy(d.initial.begin(), d.initial.end(), s.current.begin());
            std::fill(s.residue.begin(), s.residue.end(), Big{0});
            s.modulus = 1;
            const auto size = s.current.size();
            for (std::uint64_t digit = 1; digit <= cap.digits; ++digit) {
                for (std::size_t k = 0; k < size; ++k) {
                    if (!event(r, cap))
                        return;
                    s.rhs[k] = static_cast<std::uint32_t>(std::uint64_t{d.scale_inverse[k]} *
                                                          reduce(s.current[k]) % p1);
                }
                for (std::size_t k = 0; k < size; ++k)
                    for (auto pos = d.lower_offsets[k]; pos < d.lower_offsets[k + 1]; ++pos) {
                        if (!event(r, cap))
                            return;
                        const auto& entry = d.lower[pos];
                        const auto term = std::uint64_t{s.rhs[k]} * entry.value % p1;
                        s.rhs[entry.index] =
                            static_cast<std::uint32_t>((s.rhs[entry.index] + p1 - term) % p1);
                        ++r.triangular_updates;
                    }
                for (std::size_t k = size; k-- > 0;) {
                    if (!event(r, cap))
                        return;
                    s.solution[k] = static_cast<std::uint32_t>(std::uint64_t{s.rhs[k]} *
                                                               d.diagonal_inverse[k] % p1);
                    for (auto pos = d.upper_offsets[k]; pos < d.upper_offsets[k + 1]; ++pos) {
                        if (!event(r, cap))
                            return;
                        const auto& entry = d.upper[pos];
                        const auto term = std::uint64_t{s.solution[k]} * entry.value % p1;
                        s.rhs[entry.index] =
                            static_cast<std::uint32_t>((s.rhs[entry.index] + p1 - term) % p1);
                        ++r.triangular_updates;
                    }
                    s.residue[k] += s.modulus * s.solution[k];
                    r.maximum_residue_bits =
                        std::max(r.maximum_residue_bits, bit_length(s.residue[k]));
                }
                s.modulus *= p1;
                for (std::size_t row = 0; row < size; ++row) {
                    for (auto pos = d.row_offsets[row]; pos < d.row_offsets[row + 1]; ++pos) {
                        if (!event(r, cap))
                            return;
                        const auto& entry = d.matrix[pos];
                        if (s.solution[entry.col]) {
                            s.current[row] -= entry.value * s.solution[entry.col];
                            ++r.products;
                        }
                    }
                    r.maximum_predivision_bits =
                        std::max(r.maximum_predivision_bits, bit_length(s.current[row]));
                    if (s.current[row] % p1 != 0) {
                        r.status = Status::NonDivisible;
                        return;
                    }
                    s.current[row] /= p1;
                    Small magnitude = s.current[row];
                    if (magnitude < 0)
                        magnitude = -magnitude;
                    if (magnitude > d.bound[row]) {
                        r.status = Status::InvariantFailure;
                        return;
                    }
                    r.maximum_current_bits =
                        std::max(r.maximum_current_bits, bit_length(s.current[row]));
                }
                r.digits = digit;
                if ((digit & (digit - 1)) != 0)
                    continue;
                ++r.recovery_attempts;
                s.half = s.modulus / 2;
                s.recovery_bound = boost::multiprecision::sqrt(s.half);
                bool complete = true;
                for (std::size_t k = 0; k < size; ++k) {
                    if (!reconstruct(s.residue[k], s.modulus, s.recovery_bound, s.recovered[k], s,
                                     r, cap)) {
                        complete = false;
                        break;
                    }
                }
                if (r.status == Status::EventExceeded)
                    return;
                if (!complete)
                    continue;
                if (!square_certificate(d, s, r, cap)) {
                    if (r.status == Status::EventExceeded)
                        return;
                    if (r.status == Status::BitExceeded) {
                        if (digit == cap.digits)
                            return;
                        r.status = Status::DigitsExceeded;
                    }
                    continue;
                }
                for (std::size_t k = 0; k < size; ++k) {
                    if (s.recovered[k].n != 0 && d.selected[k] >= d.candidate) {
                        r.status = Status::FutureBasis;
                        return;
                    }
                    if (s.recovered[k].n != d.expected[k].n ||
                        s.recovered[k].d != d.expected[k].d) {
                        r.status = Status::ReferenceMismatch;
                        return;
                    }
                }
                r.status = Status::Success;
                return;
            }
        };
        body();
    } catch (const std::bad_alloc&) {
        r.status = Status::HeapAttempt;
    } catch (...) {
        r.status = Status::ArithmeticFailure;
    }
    bounded::watched = false;
    r.heap = bounded::heap_attempts.load();
    if (r.heap)
        r.status = Status::HeapAttempt;
    return r;
}
Data hand() {
    Matrix m;
    m.rows = 2;
    m.columns = {{{0, Q{154}}, {1, Q{77}}}, {{0, Q{77}}, {1, Q{231}}}, {{0, Q{-31}}, {1, Q{72}}}};
    const auto factor = eliminate<Mod<p1>>(m, 2, 10000, Mod<p1>::parse, true);
    Data d = prepare(m, factor, 2);
    d.expected[0] = {Big{-3}, Big{7}};
    d.expected[1] = {Big{5}, Big{11}};
    return d;
}
void controls() {
    auto d = hand();
    Scratch s(d);
    const auto bytes = payload(d, s);
    if (bytes != plan(d))
        throw std::runtime_error("hand payload forecast differs");
    const auto good = run(d, s, {bytes, 8192, 128, 1000000});
    if (good.status != Status::Success || good.heap || !good.negative_denominators)
        throw std::runtime_error("hand lift or sign normalization");
    const auto expect = [&](const Data& value, const Limits& cap, Status status) {
        Scratch temporary(value);
        const auto out = run(value, temporary, cap);
        if (out.status != status || out.heap)
            throw std::runtime_error("negative control differs");
    };
    expect(d, {bytes - 1, 8192, 128, 1000000}, Status::PayloadExceeded);
    expect(d, {bytes, 8191, 128, 1000000}, Status::BitExceeded);
    expect(d, {bytes, 8192, 0, 1000000}, Status::DigitsExceeded);
    expect(d, {bytes, 8192, 128, good.events - 1}, Status::EventExceeded);
    auto bad = d;
    bad.matrix[0].value += 1;
    expect(bad, {payload(bad, s), 8192, 128, 1000000}, Status::NonDivisible);
    bad = d;
    bad.diagonal_inverse[0] = 0;
    expect(bad, {payload(bad, s), 8192, 128, 1000000}, Status::InvalidFactor);
    bad = d;
    bad.selected[0] = d.candidate;
    expect(bad, {payload(bad, s), 8192, 128, 1000000}, Status::FutureBasis);
    bad = d;
    bad.expected[0].n = -2;
    expect(bad, {payload(bad, s), 8192, 128, 1000000}, Status::ReferenceMismatch);
    const auto repeated = run(d, s, {bytes, 8192, good.digits, good.events});
    if (repeated.status != Status::Success || repeated.events != good.events ||
        repeated.products != good.products || repeated.heap)
        throw std::runtime_error("old input or exact cap changed after failure");
    Pair value;
    Report recovery;
    Limits cap{bytes, 8192, 128, 1000000};
    bounded::heap_attempts = 0;
    bounded::watched = true;
    const bool accepted = reconstruct(Big{14}, Big{101}, Big{7}, value, s, recovery, cap);
    bounded::watched = false;
    if (!accepted || value.n != -3 || value.d != 7 || !recovery.negative_denominators ||
        bounded::heap_attempts != 0)
        throw std::runtime_error("independent signed rational literal");
    Big huge = 1;
    for (int i = 0; i < 2000; ++i)
        huge *= 10;
    s.recovered[0] = {Big{1}, huge};
    s.recovered[1] = {Big{1}, huge + 1};
    Report excessive;
    bounded::heap_attempts = 0;
    bounded::watched = true;
    const bool oversized = square_certificate(d, s, excessive, cap);
    bounded::watched = false;
    if (oversized || excessive.status != Status::BitExceeded || bounded::heap_attempts != 0)
        throw std::runtime_error("common denominator capacity rejection");
    std::cout
        << "{\"hand_coordinates\":[\"-3/7\",\"5/"
           "11\"],\"literal_modulus\":101,\"literal_residue\":14,\"negative_denominator_"
           "normalized\":true,\"payload_equal_one_short\":true,\"bits_equal_one_short\":true,"
           "\"events_equal_one_short\":true,\"digits_equal_zero\":true,\"nondivisible_rejected\":"
           "true,\"bad_pivot_rejected\":true,\"future_basis_rejected\":true,\"reference_mismatch_"
           "rejected\":true,\"common_denominator_bit_rejected\":true,\"same_input_after_failure\":"
           "true,\"heap_attempts\":0,\"hand_digits\":"
        << good.digits << ",\"hand_events\":" << good.events << "}\n";
}
} // namespace lift_fixed
int main(int argc, char** argv) try {
    using namespace lift_fixed;
    if (argc == 2 && std::string(argv[1]) == "--controls") {
        controls();
        return 0;
    }
    if (argc != 6 || !prime<p1>())
        return 2;
    const std::string domain = argv[2];
    if (domain != "full" && domain != "cut")
        return 2;
    const auto wide_candidate = std::stoull(argv[4]);
    if (wide_candidate > UINT32_MAX)
        throw std::runtime_error("candidate index overflow");
    const auto candidate = static_cast<std::uint32_t>(wide_candidate);
    const auto digits = std::stoull(argv[5]);
    std::ifstream source(argv[1]);
    const auto input = read(source);
    const auto matrix = active_matrix(input, domain == "cut", true);
    const auto factor =
        eliminate<Mod<p1>>(matrix, 33 + input.full, 200000000, Mod<p1>::parse, true);
    auto data = prepare(matrix, factor, candidate);
    expected_coordinates(data, argv[3], input.n, matrix.rows, input.full,
