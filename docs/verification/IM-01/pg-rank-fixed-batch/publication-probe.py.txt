"""Exercise public completion gate with captured results; no numeric transport replay."""
import argparse
import copy
import importlib.machinery
import importlib.util
import json
import tempfile
from pathlib import Path


def load(path):
    loader=importlib.machinery.SourceFileLoader('fixed_batch_gate',str(path))
    spec=importlib.util.spec_from_loader(loader.name,loader)
    module=importlib.util.module_from_spec(spec);loader.exec_module(module)
    return module


def main():
    p=argparse.ArgumentParser()
    p.add_argument('--runner',type=Path,required=True);p.add_argument('--capture',type=Path,required=True)
    p.add_argument('--output',type=Path,required=True);p.add_argument('--root',type=Path,required=True)
    args=p.parse_args();runner=load(args.runner)
    capture=json.loads(args.capture.read_text(encoding='utf-8'))
    first=copy.deepcopy(capture['result'])
    first.update(capture['first_probe'])
    first['columns']=[{'proof_rows':0} for _ in range(383)]
    selected={c['candidate'] for c in capture['result']['columns'] if c['selected']}
    first_index=next(j for j in range(383) if j not in selected)
    first['columns'][first_index]=copy.deepcopy(capture['result']['columns'][first_index])
    scenarios=['valid','controls-old','control-rank-float','control-flag-int','nonpivot-false',
               'nonzero-status','heap-attempt','incomplete-columns','missing-row','missing-coordinate',
               'duplicate-column','wrong-selection','digits-float','failed-column','payload-over',
               'timeout','replacement-error','swapped-input']
    outcomes=[]
    with tempfile.TemporaryDirectory() as temporary:
        directory=Path(temporary)
        for present in [False,True]:
            for name in scenarios:
                controls=copy.deepcopy(capture['controls']);all_result=copy.deepcopy(capture['result'])
                if name=='controls-old':controls.pop('hand_rank')
                elif name=='control-rank-float':controls['hand_rank']=2.0
                elif name=='control-flag-int':controls['same_scratch_after_failure']=1
                elif name=='nonpivot-false':controls['nonpivot_square_success_then_full_rejection']=False
                elif name=='nonzero-status':all_result['status']=1
                elif name=='heap-attempt':all_result['heap_attempts']=1
                elif name=='incomplete-columns':all_result['columns'].pop()
                elif name=='missing-row':all_result['columns'][-1]['proof_rows']-=1
                elif name=='missing-coordinate':all_result['columns'][-1]['coordinate_checks']-=1
                elif name=='duplicate-column':all_result['columns'][-1]['candidate']=0
                elif name=='wrong-selection':
                    a=next(c for c in all_result['columns'] if c['selected'])
                    b=next(c for c in all_result['columns'] if not c['selected'])
                    a['selected']=False;b['selected']=True
                elif name=='digits-float':all_result['columns'][-1]['digits']=1.0
                elif name=='failed-column':all_result['failed_column']=382
                elif name=='payload-over':all_result['payload_bytes']=48*1024**3+1
                output=directory/'result.json';output.unlink(missing_ok=True)
                if present:output.write_bytes(b'prior completed result\n')
                calls=[]
                def invoke(executable,*values):
                    calls.append(str(values[-1]))
                    if name=='timeout' and values[-1]=='all':raise TimeoutError('injected timeout')
                    if values==('--controls',):return controls
                    return first if values[-1]=='first' else all_result
                runner.invoke=invoke
                old_replace=runner.os.replace
                def fail_replace(*unused):raise OSError('injected replacement failure')
                if name=='replacement-error':runner.os.replace=fail_replace
                inputs=args.root/'out/issue140-scratch'
                if name=='swapped-input':
                    inputs=directory/'inputs';inputs.mkdir(exist_ok=True)
                    (inputs/'1-quartic.txt').write_bytes((args.root/'out/issue140-scratch/1-flat.txt').read_bytes())
                argv=['--inputs',str(inputs),'--certificates',str(args.root/'out/issue164-fixed-verifier'),
                      '--batch',str(args.root/'out/issue140-fixed-batch/batch.exe'),
                      '--capacity-producer',str(args.root/'docs/verification/IM-01/pg-rank-fixed-slot-design/capacity-oracle-v2.py.txt'),
                      '--study',str(args.root/'docs/verification/IM-01/pg-rank-resource-study/results.json'),
                      '--certificate-index',str(args.root/'docs/verification/IM-01/pg-rank-fixed-certificate/certificate-index.json'),
                      '--output',str(output)]
                accepted=False
                try:runner.main(argv);accepted=True
                except (RuntimeError,ValueError,TimeoutError,OSError):pass
                finally:runner.os.replace=old_replace
                runner.require(accepted==(name=='valid'),f'{name}: public completion gate')
                if accepted:
                    runner.require(json.loads(output.read_text())['result']['coordinate_checks']==70089,'normal complete publication')
                else:
                    runner.require(output.read_bytes()==b'prior completed result\n' if present else not output.exists(),'old output retained')
                runner.require(not list(directory.glob('.result.json.*.tmp')),'temporary cleanup')
                if name.startswith('control') or name=='nonpivot-false':
                    runner.require(calls==['--controls'],'controls rejected before numeric transport')
                if name=='swapped-input':runner.require(not calls,'identity rejected before transport')
                outcomes.append({'scenario':name,'previous_output':present,'accepted':accepted,'transport_calls':len(calls)})
    args.output.write_text(json.dumps({'schema':'uvlas.fixed-batch-publication-probe/1','numeric_transport':'captured-result stub',
                                      'cases':outcomes},indent=2)+'\n',encoding='utf-8',newline='\n')


if __name__=='__main__':main()
