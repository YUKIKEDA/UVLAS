"""Validate the n4 full-background fixed batch against the frozen certificate."""

import argparse
import hashlib
import json
from pathlib import Path
from runpy import run_path


SHAPES = {
    'flat': (3904, 674, 1237, 490),
    'quartic': (20032, 933, 1496, 774),
}
EXPECTED_DRIVER = {
    'schema': 'uvlas.fixed-batch-n4-full-driver/1',
    'base_sha256': 'f46f8dbe506c1efa7e834c298b64dc1329c0d4b6b96ff1cb893ade44ce7cdb63',
    'generated_source_sha256': '1098f4a3e7f9ee2666c94c527fe89350415c4fe7a02f78113adf4011a5243526',
    'generated_bytes': 38327,
    'changed_scope': 'main n guard and diagnostic only',
    'kernel_bytes_preserved': True,
}


def sha256(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def validate_driver(base, record):
    base.require(type(record) is dict, 'driver record object')
    for key, expected in EXPECTED_DRIVER.items():
        value = record.get(key)
        base.require(type(value) is type(expected) and value == expected, f'driver {key} identity')


def validate_batch(base, result, dimensions, selected, first):
    rows, _, candidates, rank = dimensions
    require = base.require
    integer = base.integer
    require(type(result) is dict, 'result object')
    columns_checked = 1 if first else candidates
    for key, expected in {
        'status': 0, 'rows': rows, 'candidates': candidates, 'rank': rank, 'heap_attempts': 0,
        'failed_column': 2**32 - 1, 'selected': 0 if first else rank,
        'dependent': 1 if first else candidates - rank, 'proof_rows': columns_checked * rows,
        'coordinate_checks': columns_checked * rank,
    }.items():
        integer(result.get(key), expected, key)
    for key, low, high in (
        ('payload_bytes', 1, 48 * 1024**3), ('defined_events', 1, 2_000_000_000),
        ('integer_bits', 1, 1536), ('proof_products', 0, 2**64 - 1),
        ('batch_ms', 0, 2**64 - 1), ('process_peak_bytes', 1, 2**64 - 1),
    ):
        value = result.get(key)
        require(type(value) is int and low <= value <= high, f'{key} range')
    columns = result.get('columns')
    require(type(columns) is list and len(columns) == candidates, 'column roster')
    require(all(type(column) is dict for column in columns), 'column record object')
    active = [column for column in columns if column.get('proof_rows')] if first else columns
    require(len(active) == columns_checked, 'processed column count')
    ordinals = [next(j for j in range(candidates) if j not in selected)] if first else range(candidates)
    for j, column in zip(ordinals, active, strict=True):
        integer(column.get('candidate'), j, 'candidate order')
        integer(column.get('lift_status'), 0, 'lift status')
        integer(column.get('proof_rows'), rows, 'column row completeness')
        integer(column.get('coordinate_checks'), rank, 'column coordinate completeness')
        require(column.get('selected') is (j in selected), 'certificate selection')
        base.validate_lift_metrics(column, result['defined_events'])
        digits = column.get('digits')
        schedule = (0,) if column['selected'] else (1, 2, 4, 8, 16, 32, 64, 128)
        require(type(digits) is int and digits in schedule, 'digit schedule')
        width = column.get('proof_bound_bits')
        require(type(width) is int and 0 < width <= 8192, 'proof width')
        products = column.get('proof_products')
        require(type(products) is int and products >= 0, 'proof products type')
    require(sum(c['defined_lift_events'] for c in active) <= result['defined_events'], 'lift event total')
    integer(sum(c['proof_products'] for c in active), result['proof_products'], 'proof product total')


def read_basis(base, certificate, dimensions):
    _, _, candidates, rank = dimensions
    with certificate.open(encoding='ascii') as stream:
        header = stream.readline().split()
        base.require(header == ['certificate', '4', *map(str, dimensions)], 'certificate header')
        selected = set()
        for i in range(rank):
            fields = stream.readline().split()
            base.require(len(fields) == 4 and fields[:2] == ['basis', str(i)], 'certificate basis row')
            candidate = int(fields[2])
            base.require(0 <= candidate < candidates and candidate not in selected, 'certificate basis candidate')
            selected.add(candidate)
    return selected


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--shape', choices=SHAPES, required=True)
    for name in (
        'inputs', 'certificates', 'batch', 'base-runner', 'shared-runner',
        'capacity-producer', 'study', 'certificate-index', 'driver-record', 'output',
    ):
        parser.add_argument(f'--{name}', type=Path, required=True)
    args = parser.parse_args(argv)
    shared = run_path(str(args.shared_runner))
    reject_aliases = shared['reject_output_aliases']
    sources = {shape: args.inputs / f'4-{shape}.txt' for shape in SHAPES}
    reject_aliases(args.output, (
        *sources.values(), args.batch, args.base_runner, args.shared_runner, args.capacity_producer,
        args.study, args.certificate_index, args.driver_record,
    ))
    base = shared['load']('fixed_batch_base', args.base_runner)
    capacity = shared['load']('fixed_batch_capacity', args.capacity_producer)
    driver = json.loads(args.driver_record.read_bytes())
    validate_driver(base, driver)
    index = json.loads(args.certificate_index.read_bytes())['certificates']
    records = {}
    for shape in SHAPES:
        rows = [
            row for row in index
            if (row['n'], row['shape'], row['domain'], row['row_order']) == (4, shape, 'full', 'sparse')
        ]
        base.require(len(rows) == 1, f'unique certificate for {shape}')
        records[shape] = rows[0]
    certificates = {shape: args.certificates / row['file'] for shape, row in records.items()}
    reject_aliases(args.output, certificates.values())
    source = sources[args.shape]
    record = records[args.shape]
    certificate = certificates[args.shape]
    dimensions = SHAPES[args.shape]
    rank = dimensions[-1]
    study = json.loads(args.study.read_bytes())['cases']
    prior = capacity.verify_input(source, 4, args.shape, 'full', study)
    base.require((prior['rows'], prior['columns'], prior['candidates'], prior['mod_rank']) == dimensions,
                 'study dimensions')
    base.require(record['rank'] == rank and record['input_sha256'] == prior['input_sha256'],
                 'coefficient and certificate identity')
    base.require(certificate.stat().st_size == record['bytes'] and sha256(certificate) == record['sha256'],
                 'certificate identity')
    selected = read_basis(base, certificate, dimensions)
    binary = args.batch.resolve()
    controls = base.invoke(binary, '--controls')
    base.validate_controls(controls)
    first = base.invoke(binary, source, certificate, 'first')
    validate_batch(base, first, dimensions, selected, True)
    result = base.invoke(binary, source, certificate, 'all')
    validate_batch(base, result, dimensions, selected, False)
    base.publish(args.output, {
        'schema': 'uvlas.fixed-batch-n4-full/1', 'n': 4, 'shape': args.shape,
        'domain': 'full', 'row_order': 'sparse',
        'input_sha256': record['input_sha256'], 'certificate_sha256': record['sha256'],
        'certificate_bytes': record['bytes'], 'binary_sha256': sha256(binary),
        'driver': driver, 'controls': controls,
        'first_probe': {key: value for key, value in first.items() if key != 'columns'},
        'result': shared['compact'](result),
    })


if __name__ == '__main__':
    main()
