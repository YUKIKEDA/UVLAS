ExactRational owned(Fixture& f, const Q& value) {
    return rat(f, numerator(value), denominator(value));
}

struct RuleNode {
    ExactRational u, v, depth;
};

std::vector<RuleNode> finite_rule(Fixture& f) {
    // Finite rule solely for the geometry inner-product certificate of this fixture.
    // The R4 mechanics quadrature is not used by the trace operation.
    constexpr unsigned order = 10;
    const std::array<Q, 3> depths{Q{1, 4}, Q{1, 2}, Q{3, 4}};
    std::vector<RuleNode> nodes;
    nodes.reserve(3 * (order + 1) * (order + 2) / 2);
    for (unsigned i = 0; i <= order; ++i)
        for (unsigned j = 0; j <= order - i; ++j) {
            const Q u = Q{512 * (i + 1) + (j + 1) * (j + 2), 512 * (order + 3)};
            const Q v = Q{512 * (j + 1) + (i + 1) * (i + 3), 512 * (order + 3)};
            REQUIRE(u > 0);
            REQUIRE(v > 0);
            REQUIRE(u + v < 1);
            for (const auto& depth : depths)
                nodes.push_back({owned(f, u), owned(f, v), owned(f, depth)});
        }
    return nodes;
}

constexpr std::uint64_t sample_prime = 1'000'000'007;

std::uint64_t sample_power(std::uint64_t base, std::uint64_t exponent) {
    std::uint64_t result = 1;
    while (exponent != 0) {
        if ((exponent & 1U) != 0)
            result = result * base % sample_prime;
        exponent >>= 1U;
        base = base * base % sample_prime;
    }
    return result;
}

std::uint64_t residue(const Q& value) {
    I numerator_units = numerator(value) % sample_prime;
    if (numerator_units < 0)
        numerator_units += sample_prime;
    const I denominator_units = denominator(value) % sample_prime;
    REQUIRE(denominator_units != 0);
    return numerator_units.convert_to<std::uint64_t>() *
           sample_power(denominator_units.convert_to<std::uint64_t>(), sample_prime - 2) %
           sample_prime;
}

struct SampleTerm {
    std::size_t column, field;
    std::uint64_t x, y;
    std::uint64_t value;
};

class SampleRank {
public:
    SampleRank(const PgRankPreparation& basis, std::size_t columns, std::size_t extent,
               core::ResourcePlan& plan)
        : terms_(extent * extent), pivots_(columns), columns_(columns), extent_(extent) {
        const auto rows = take(basis.rows());
        const auto exact = query_limits().algebra;
        for (std::size_t col = 0; col < columns; ++col) {
            const PgRankBasisColumn column{static_cast<core::Index>(col)};
            for (const auto i : take(basis.basis_indices(column))) {
                const auto& row = rows[i];
                const auto value =
                    take(basis.basis(column, static_cast<core::Index>(i), exact, plan));
                terms_[cell(row.cell)].push_back({col, static_cast<std::size_t>(row.field),
                                                  static_cast<std::uint64_t>(row.powers.first),
                                                  static_cast<std::uint64_t>(row.powers.second),
                                                  residue(q(value))});
            }
        }
    }

    [[nodiscard]] bool add(const PgCellIndex& index, const Q& s, const Q& t) {
        if (rank_ == columns_)
            return false;
        const auto x = residue(2 * extent_ * s - 2 * index.first - 1);
        const auto y = residue(2 * extent_ * t - 2 * index.second - 1);
        std::array<std::vector<std::uint64_t>, 3> values;
        for (auto& field : values)
            field.resize(columns_);
        for (const auto& term : terms_[cell(index)]) {
            auto& value = values[term.field][term.column];
            value = (value + term.value * sample_power(x, term.x) % sample_prime *
                                 sample_power(y, term.y) % sample_prime) %
                    sample_prime;
        }
        const auto before = rank_;
        for (auto& field : values)
            add_row(std::move(field));
        return rank_ != before;
    }

    [[nodiscard]] std::size_t rank() const noexcept {
        return rank_;
    }

private:
    [[nodiscard]] std::size_t cell(const PgCellIndex& index) const {
        return static_cast<std::size_t>(index.first) * extent_ +
               static_cast<std::size_t>(index.second);
    }

    void add_row(std::vector<std::uint64_t> row) {
        for (std::size_t i = 0; i < row.size(); ++i) {
            if (row[i] == 0)
                continue;
            if (pivots_[i].empty()) {
                const auto inverse = sample_power(row[i], sample_prime - 2);
                for (std::size_t j = i; j < row.size(); ++j)
                    row[j] = row[j] * inverse % sample_prime;
                pivots_[i] = std::move(row);
                ++rank_;
                return;
            }
            const auto scale = row[i];
            for (std::size_t j = i; j < row.size(); ++j)
                row[j] =
                    (row[j] + sample_prime - scale * pivots_[i][j] % sample_prime) % sample_prime;
        }
    }

    std::vector<std::vector<SampleTerm>> terms_;
    std::vector<std::vector<std::uint64_t>> pivots_;
    std::size_t columns_, extent_, rank_{};
};

std::vector<PgDiscreteVolumePoint> select_rule(Fixture& f, const PgReferenceBasis& basis,
                                               core::Index n, std::span<const RuleNode> nodes,
                                               std::vector<ExactRational>& weights,
                                               core::ResourcePlan& plan) {
    const auto& partition = take(basis.source()).partition;
    const auto volumes = take(partition.volumes());
    const auto triangles = take(partition.triangles());
    const auto vertices = take(partition.vertices());
    const auto pieces = take(partition.pieces());
    const auto columns = static_cast<std::size_t>(take(basis.size()));
    SampleRank selection{*take(basis.rank()), columns, 2 * static_cast<std::size_t>(n), plan};
    std::vector<std::vector<std::size_t>> selected(volumes.size());
    constexpr std::size_t stride = 3;
    for (std::size_t i = 0; i < volumes.size(); ++i) {
        const auto& tri = triangles[static_cast<std::size_t>(volumes[i].triangle)];
        const auto& a = vertices[static_cast<std::size_t>(tri.vertices[0])];
        const auto& b = vertices[static_cast<std::size_t>(tri.vertices[1])];
        const auto& c = vertices[static_cast<std::size_t>(tri.vertices[2])];
        const auto& index = pieces[static_cast<std::size_t>(tri.piece)].cell;
        const auto material_coordinate = [&](const RuleNode& p, auto member) -> Q {
            return q(a.*member) + q(p.u) * (q(b.*member) - q(a.*member)) +
                   q(p.v) * (q(c.*member) - q(a.*member));
        };
        for (std::size_t j = 0; j < nodes.size(); j += stride) {
            const auto& p = nodes[j];
            const auto gain = selection.add(index, material_coordinate(p, &PgPartitionPoint::s),
                                            material_coordinate(p, &PgPartitionPoint::t));
            if (gain || j == 0)
                selected[i].push_back(j);
        }
    }
    // Three distinct depths at each point separate the three field powers.
    // The finite-field sweep only chooses positive test samples. The normal
    // product's interval factor and congruence checks still decide acceptance.
    CAPTURE(n, columns, selection.rank());
    REQUIRE(selection.rank() == columns);
    weights.reserve(volumes.size());
    std::vector<PgDiscreteVolumePoint> points;
    for (std::size_t i = 0; i < volumes.size(); ++i) {
        const auto& weight =
            weights.emplace_back(owned(f, Q{1} / (2 * stride * selected[i].size())));
        for (const auto j : selected[i])
            for (const auto& node : nodes.subspan(j, stride))
                points.push_back(
                    {{static_cast<core::Index>(i), node.u, node.v, node.depth}, weight});
    }
    return points;
}

void set_fixture(Fixture& fixture, const Input& input, core::Index n, const Q& h0) {
    const Q skew(input.skew);
    std::array<std::array<Q, 5>, 5> expanded{};
    for (std::size_t ordinal = 0; ordinal < exponents.size(); ++ordinal) {
        if (input.coefficients[ordinal].empty())
            continue;
        const Q coefficient(input.coefficients[ordinal]);
        const auto [px, py] = exponents[ordinal];
        Q skew_power(1);
        unsigned choose = 1;
        for (unsigned j = 0; j <= px; ++j) {
            expanded[px - j][py + j] += coefficient * choose * skew_power;
            skew_power *= skew;
            if (j < px)
                choose = choose * (px - j) / (j + 1);
        }
    }
    std::array<std::array<ExactRational, 5>, 5> owned_coefficients;
    std::vector<ExactInputTerm> terms;
    for (unsigned i = 0; i < 5; ++i)
        for (unsigned j = 0; j < 5 - i; ++j) {
            if (expanded[i][j] == 0)
                continue;
            auto& value = owned_coefficients[i][j];
            value = owned(fixture, expanded[i][j]);
            terms.push_back({{i, j}, value});
        }
    auto graph = take(ExactChartPolynomial::make(terms, exact_limits(), fixture.input_plan));
    const auto zero = fixture.r("0"), one = fixture.r("1");
    fixture.graph = take(
        PgGraph::prepare({zero, one, owned(fixture, skew), graph, n, PgGraphConvention::Version1},
                         {exact_limits(), 64, {}}, fixture.input_plan));
    if (input.cut) {
        fixture.domain.clear();
        fixture.domain.push_back(fixture.point("0", "0"));
        fixture.domain.push_back({fixture.r("1"), fixture.r("3", "10")});
        fixture.domain.push_back({fixture.r("1"), fixture.r("9", "10")});
        fixture.domain.push_back(fixture.point("0", "1"));
    }
    for (std::size_t i = 0; i < fixture.heights.size(); ++i) {
        const auto& vertex = fixture.thickness[i];
        const Q factor = input.tapered ? Q(1 + q(vertex.s) / 10 + q(vertex.t) / 20) : Q(1);
        fixture.heights[i] = owned(fixture, h0 * factor);
    }
    fixture.layers.clear();
    fixture.layers.push_back({core::LayerId{1}, fixture.zeta.front(), fixture.zeta.back()});
}

std::uint64_t check_fields(const PgConditioningCertificate& certificate,
                           const PgConditioningInfo& info, const PgReferenceLimits& arithmetic,
                           core::Workspace& workspace, core::ResourcePlan& audit_plan) {
    const auto* rank = take(take(certificate.reference_basis())->rank());
    const auto n = static_cast<std::size_t>(info.rank);
    const auto prefix = static_cast<std::size_t>(info.protected_rank);
    REQUIRE((prefix != 0 && prefix <= 33));
    std::vector<std::vector<Q>> transform(prefix, std::vector<Q>(prefix));
    for (std::size_t j = 0; j < prefix; ++j)
        for (std::size_t i = 0; i <= j; ++i)
            transform[i][j] = q(take(certificate.transform_coefficient(
                {static_cast<core::Index>(i)}, {static_cast<core::Index>(j)}, arithmetic.algebra,
                audit_plan)));
    std::uint64_t comparisons = 0;
    for (core::Index field = 0; field < 33; ++field) {
        const auto coordinates =
            take(certificate.required_coordinates({field}, arithmetic, workspace, audit_plan));
        REQUIRE(static_cast<std::size_t>(coordinates.size()) == n);
        for (std::size_t i = 0; i < n; ++i) {
            if (i >= prefix)
                REQUIRE(coordinates.values()[i].sign() == 0);
            Q original{};
            for (std::size_t j = i; j < prefix; ++j)
                original += transform[i][j] * q(coordinates.values()[j]);
            const auto expected = take(rank->required({static_cast<core::Index>(i)}, {field},
                                                      arithmetic.algebra, audit_plan));
            REQUIRE(original == q(expected));
            ++comparisons;
        }
    }
    return comparisons;
}

std::uint64_t peak_resident_bytes() {
#ifdef _WIN32
    PROCESS_MEMORY_COUNTERS value{};
    REQUIRE(GetProcessMemoryInfo(GetCurrentProcess(), &value, static_cast<DWORD>(sizeof(value))) !=
            0);
    return value.PeakWorkingSetSize;
#else
    rusage value{};
    REQUIRE(getrusage(RUSAGE_SELF, &value) == 0);
    REQUIRE(value.ru_maxrss > 0);
    return 1024U * static_cast<std::uint64_t>(value.ru_maxrss);
#endif
}

void verify_case(const Input& input, core::Index n, const char* thickness) {
    using Clock = std::chrono::steady_clock;
    const auto start = Clock::now();
    const auto ms = [](auto elapsed) {
        return std::chrono::duration_cast<std::chrono::milliseconds>(elapsed).count();
    };
    CAPTURE(input.name, n, thickness);
    Fixture f;
    set_fixture(f, input, n, Q(thickness));
    const auto one = f.r("1");
    std::array<ExactRational, 9> rotation;
    for (const auto i : {0U, 4U, 8U})
        rotation[i] = f.r("1");
    const std::array<ExactRational, 3> origin;
    core::ResourcePlan plan{profile}, audit_plan{profile};
    core::Workspace workspace{core::PlannedBuffer<std::byte>{}};
    auto basis = make_basis(f, {rotation, origin, one}, workspace, plan);
    const auto n_basis = take(basis.size());
    const auto n_volumes = take(take(basis.source()).partition.volumes()).size();
    const auto basis_charged = plan.allocated_bytes();
    const auto nodes = finite_rule(f);
    std::vector<ExactRational> weights;
    const auto points = select_rule(f, basis, n, nodes, weights, audit_plan);
    std::clog << "VPG_INPUT," << input.name << ',' << n << ',' << thickness << ',' << n_basis << ','
              << n_volumes << ',' << points.size() << ',' << basis_charged << ','
              << audit_plan.allocated_bytes() << '\n'
              << std::flush;
    auto arithmetic = query_limits();
    arithmetic.maximum_scalar_operations = 1'000'000'000;
    arithmetic.maximum_partition_visits = 300'000'000;
    const auto begin_gram = Clock::now();
    auto enclosure = PgDiscreteGram::prepare(std::move(basis), points, {768, 704},
                                             {arithmetic, steps}, workspace, plan);
    if (!enclosure) {
        std::clog << "VPG_REJECT,gram," << input.name << ',' << n << ',' << thickness << ','
                  << static_cast<unsigned>(enclosure.error().code) << ',' << plan.allocated_bytes()
                  << ',' << ms(Clock::now() - begin_gram) << '\n'
                  << std::flush;
        // NOLINTNEXTLINE(bugprone-use-after-move): prepare transfers ownership only on success.
        CHECK(take(basis.size()) == n_basis);
        FAIL("Gram preparation was refused");
    }
    const auto gram_info = enclosure->info();
    const auto gram_charged = plan.allocated_bytes();
    const auto begin_certificate = Clock::now();
    std::clog << "VPG_GRAM," << input.name << ',' << n << ',' << thickness << ','
              << gram_info.integer_steps << ',' << gram_info.arithmetic.scalar_operations << ','
              << gram_info.arithmetic.partition_visits << ',' << gram_charged << ','
              << ms(begin_certificate - begin_gram) << '\n'
              << std::flush;
    auto qualified = std::move(*enclosure)
                         .generate_and_certify({768, 128}, {profile, steps, {}}, workspace, plan);
    if (!qualified) {
        std::clog << "VPG_REJECT,conditioning," << input.name << ',' << n << ',' << thickness << ','
                  << static_cast<unsigned>(qualified.error().code) << ',' << plan.allocated_bytes()
                  << ',' << ms(Clock::now() - begin_certificate) << '\n'
                  << std::flush;
        CHECK(take(enclosure->enclosure()).upper_triangle.size() ==
              static_cast<std::size_t>(n_basis * (n_basis + 1) / 2));
        FAIL("Finite-rule qualification was refused");
    }
    const auto end_certificate = Clock::now();
    const auto& certificate = qualified->conditioning();
    const auto info = take(certificate.info());
    const auto bound = q(take(certificate.congruence_bound(arithmetic.algebra, audit_plan)));
    REQUIRE((bound >= 0 && bound <= Q("1/100000000000000000000")));
    REQUIRE(info.rank == static_cast<std::uint64_t>(n_basis));
    const auto owned_points = qualified->rule();
    REQUIRE(owned_points.size() == points.size());
    for (std::size_t i = 0; i < points.size(); ++i) {
        const auto& chosen = points[i];
        const auto& stored = owned_points[i];
        REQUIRE(stored.volume == chosen.point.volume);
        REQUIRE(stored.u == chosen.point.u);
        REQUIRE(stored.v == chosen.point.v);
        REQUIRE(stored.depth == chosen.point.depth);
        REQUIRE(stored.weight == chosen.weight);
    }
    arithmetic.algebra.maximum_value_bits = 65'536;
    arithmetic.algebra.maximum_work_bits = 262'144;
    const auto mandatory = check_fields(certificate, info, arithmetic, workspace, audit_plan);
    REQUIRE(mandatory == 33 * info.rank);
    std::clog << "VPG_ACCEPT," << input.name << ',' << n << ',' << thickness << ',' << bound << ','
              << info.protected_rank << ',' << info.steps << ',' << info.payload_bytes << ','
              << plan.allocated_bytes() << ',' << ms(end_certificate - begin_certificate) << ','
              << ms(end_certificate - start) << ',' << mandatory << ','
              << audit_plan.allocated_bytes() << ',' << peak_resident_bytes() << '\n'
              << std::flush;
}

} // namespace

TEST_CASE("PG qualifies finite rules for the frozen R4 and R5 shape series",
          "[SH-D03][SH-D07][N4-02][N4-08][V-PG][verification]") {
    for (const auto& input : inputs)
        for (const auto n : {input.coarse, 2 * input.coarse})
            for (const auto* thickness : {"1/100", "1/10000"}) {
                DYNAMIC_SECTION(input.name << "_n" << n << "_h" << thickness) {
                    verify_case(input, n, thickness);
                }
            }
}
