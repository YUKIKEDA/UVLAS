void measure_candidate(std::size_t patches, std::size_t contact_count, PgSparseChart&& chart,
                       const PgGlobalConstraintInfo& info, const PgReferenceLimits& limits,
                       core::Workspace& workspace, core::ResourcePlan& plan) {
    using Buffer = numerics::PrecisionBuffer<double>;
    const auto rank = static_cast<core::Index>(chart.pivots().size());
    const auto cols = static_cast<std::size_t>(info.columns);
    const auto rows = static_cast<std::size_t>(info.rows);
    const auto count = chart.free_columns().size();
    const auto identity = take(chart.identity());
    sparse_test::Fixture forward{rank}, transpose{rank};
    auto transposed = sparse_test::context();
    transposed.unknown_layout = core::ScopeId{511};
    transposed.residual_layout = core::ScopeId{512};
    const auto direct = forward.scales(), reverse = transpose.scales(transposed);
    const auto solver = sparse_test::limits(rank, 3);
    const auto make = [&](std::size_t n, std::size_t offset, bool zero) {
        std::vector<double> values(3 * n);
        if (!zero)
            for (std::size_t c = 0; c < 3; ++c)
                for (std::size_t i = 0; i < n; ++i)
                    values[c * n + i] =
                        static_cast<double>(static_cast<int>((i + 2 * c + offset) % 9) - 4);
        const auto index = static_cast<core::Index>(n);
        auto dense = take(numerics::DenseReal::copy({index, 3, index}, values, plan,
                                                    core::Operation::EvaluateStructure));
        return take(Buffer::binary64(std::move(dense), core::NumericalGeneration{91}));
    };
    const auto r = make(count, 1, false);
    const auto f = make(cols, 3, false);
    const auto c = make(rows, 0, true);
    const auto request = [&](const numerics::linear::Scales& scales) {
        return PgChartCandidateSolve{scales,
                                     forward.precision(),
                                     sparse_test::accuracy(),
                                     solver,
                                     core::RhsGeneration{88},
                                     core::NumericalGeneration{90},
                                     core::NumericalGeneration{110},
                                     1e-5,
                                     2'000'000,
                                     {}};
    };

    auto bytes = plan.allocated_bytes();
    auto time = Clock::now();
    auto candidate = PgChartCandidate::prepare(std::move(chart), direct, reverse,
                                               {limits, 0x1p-30, solver}, plan);
    const auto projected = finish(time, plan, bytes);
    std::cout << "R02-CANDIDATE {\"patches\":" << patches << ",\"connections\":" << contact_count
              << ",\"rank\":" << rank;
    write_phase("project_and_factor", projected);
    if (!candidate) {
        std::cout << ",\"prepare_failure\":" << static_cast<unsigned>(candidate.error().code)
                  << "}\n";
        return;
    }
    bytes = plan.allocated_bytes();
    time = Clock::now();
    const auto q = candidate->lift({identity, r}, {identity, c}, request(direct), workspace, plan);
    const auto lift = finish(time, plan, bytes);
    bytes = plan.allocated_bytes();
    time = Clock::now();
    const auto dual = candidate->transpose({identity, f}, request(reverse), workspace, plan);
    const auto transposed_phase = finish(time, plan, bytes);
    write_phase("lift", lift);
    write_phase("transpose", transposed_phase);
    if (q)
        std::cout << ",\"residual_upper\":" << q->original_residual_upper;
    else
        std::cout << ",\"lift_failure\":" << static_cast<unsigned>(q.error().code)
                  << ",\"lift_operation\":" << static_cast<unsigned>(q.error().operation);
    if (dual)
        std::cout << ",\"balance_upper\":" << dual->original_balance_upper;
    else
        std::cout << ",\"transpose_failure\":" << static_cast<unsigned>(dual.error().code)
                  << ",\"transpose_operation\":" << static_cast<unsigned>(dual.error().operation);
    if (q && dual) {
        time = Clock::now();
        const auto work = candidate->work_defect({identity, f}, {identity, r}, {identity, c}, *q,
                                                 *dual, 1e-6, 10'000, {});
        write_phase("work", finish(time, plan, plan.allocated_bytes()));
        if (work)
            std::cout << ",\"work_defect_upper\":" << *work;
        else
            std::cout << ",\"work_failure\":" << static_cast<unsigned>(work.error().code);
    }
    std::cout << "}\n";
}
