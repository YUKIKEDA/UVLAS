"""Run the product rank preparation and independent reference for all twelve cases."""

import argparse
import hashlib
import json
import os
import subprocess
import tempfile
from pathlib import Path

INDEX_SHA256 = "7f5168f8a8f39d8c9a6e908283cd8d6f0e4f4c11816ec1a8e1e31edb395f6b5c"
RANK_CAP = 48 * 1024**3
STEP_CAP = 10_000_000_000_000
DIGIT_CAP = 128
PLANNED_CPP_ALLOCATIONS = 44
GRIDS = (1, 2, 4)
SHAPES = ("flat", "quartic")
DOMAINS = ("full", "cut")
CASES = tuple((n, shape, domain) for n in GRIDS for shape in SHAPES for domain in DOMAINS)


def require(condition, message):
    if not condition:
        raise ValueError(message)


def digest(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def protect_inputs(output, inputs):
    for path in inputs:
        alias = output.resolve() == path.resolve() or (
            output.exists() and output.samefile(path)
        )
        require(not alias, f"output aliases an input: {path.name}")


def atomic_write(output, result):
    record = json.dumps(result, sort_keys=True, indent=2).encode("utf-8") + b"\n"
    with tempfile.NamedTemporaryFile(dir=output.parent, delete=False) as temp:
        name = Path(temp.name)
        try:
            temp.write(record)
            temp.flush()
            os.fsync(temp.fileno())
            temp.close()
            os.replace(name, output)
        finally:
            name.unlink(missing_ok=True)


def verify_result(report, case, certificate):
    require(type(report) is dict, f"native report: {case}")
    n, shape, domain = case
    require((report.get("n"), report.get("shape"), report.get("domain")) == case, f"case {case}")
    fields = (
        "active_cells", "active_rows", "anchor_rows", "candidates", "normalized_terms", "rank",
        "rank_c", "independent_required", "proof_rows", "verified_anchor_rows", "steps", "prime",
        "rank_payload_forecast", "rank_payload_actual", "pre_rank_plan_bytes", "combined_plan_bytes",
        "rank_cpp_allocation_attempts", "planned_cpp_allocations", "rank_milliseconds",
        "frozen_dictionary_terms", "frozen_required_terms", "frozen_anchor_terms",
        "coordinate_comparisons", "nonzero_coordinate_comparisons", "source_coordinate_comparisons",
        "basis_coefficient_comparisons", "oracle_plan_bytes",
        "graph_external_estimate", "dictionary_external_estimate", "partition_external_estimate",
        "rank_payload_limit", "combined_forecast_bytes", "product_peak_resident_bytes",
        "witness_domain_vertices", "witness_activity_cells", "witness_height_terms",
    )
    require(all(type(report.get(field)) is int and report[field] >= 0 for field in fields),
            f"reported fields: {case}")
    require(
        report["rank_cpp_allocation_attempts"] == report["planned_cpp_allocations"]
        == PLANNED_CPP_ALLOCATIONS,
        f"audited rank allocations: {case}",
    )
    require(0 < report["active_cells"] <= (2 * n)**2 and report["rank"] == certificate["rank"],
            f"activity and rank: {case}")
    require(
        report["witness_domain_vertices"] == (8 if domain == "cut" else 4)
        and report["witness_activity_cells"] == (2 * n)**2
        and report["witness_height_terms"] == (15 if shape == "quartic" else 0),
        f"owned source witness: {case}",
    )
    require(
        0 < report["nonzero_coordinate_comparisons"] <= report["coordinate_comparisons"]
        == report["candidates"] * report["rank"],
        f"coordinate coverage: {case}",
    )
    require(
        report["source_coordinate_comparisons"] > 0
        and report["basis_coefficient_comparisons"] == report["active_rows"] * report["rank"],
        f"basis and source coverage: {case}",
    )
    phase_estimates = sum(report[key] for key in (
        "graph_external_estimate", "dictionary_external_estimate", "partition_external_estimate"
    ))
    prior = report["pre_rank_plan_bytes"] + phase_estimates
    require(
        report["rank_payload_limit"] == RANK_CAP - prior
        and report["combined_forecast_bytes"] == prior + report["rank_payload_forecast"]
        and 0 < report["product_peak_resident_bytes"] <= RANK_CAP,
        f"combined envelope and product process peak: {case}",
    )
    require(
        report["combined_plan_bytes"] == report["pre_rank_plan_bytes"] + report["rank_payload_actual"]
        and report["combined_plan_bytes"] <= RANK_CAP
        and 0 < report["rank_payload_actual"] <= report["rank_payload_forecast"]
        <= report["rank_payload_limit"]
        and 0 < report["steps"] <= STEP_CAP,
        f"single product plan: {case}",
    )


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--binary", required=True, type=Path)
    parser.add_argument("--index", required=True, type=Path)
    parser.add_argument("--coefficient-dir", required=True, type=Path)
    parser.add_argument("--certificate-dir", required=True, type=Path)
    parser.add_argument("--output", required=True, type=Path)
    parser.add_argument("--timeout-seconds", type=int, default=900)
    args = parser.parse_args()
    require(args.timeout_seconds > 0, "positive per-case timeout required")

    require(digest(args.index) == INDEX_SHA256, "frozen index SHA-256 mismatch")
    index = json.loads(args.index.read_bytes())
    certificates = {
        (row["n"], row["shape"], row["domain"]): row for row in index["certificates"]
        if row["row_order"] == "sparse"
    }
    require(len(index["certificates"]) == len(CASES) and set(certificates) == set(CASES),
            "frozen case roster")
    pending = {}
    inputs = {args.binary, args.index}
    source_hashes = {}
    for case in CASES:
        n, shape, domain = case
        entry = certificates[case]
        label = f"{n}-{shape}-{domain}"
        coefficient = args.coefficient_dir / f"{n}-{shape}.txt"
        certificate = args.certificate_dir / f"{label}.cert"
        require(certificate.name == entry["file"], f"certificate name: {label}")
        inputs.update((coefficient, certificate))
        if coefficient not in source_hashes:
            source_hashes[coefficient] = digest(coefficient)
        require(source_hashes[coefficient] == entry["input_sha256"], f"coefficient SHA-256: {label}")
        require(certificate.stat().st_size == entry["bytes"] and digest(certificate) == entry["sha256"],
                f"certificate identity: {label}")
        pending[case] = (coefficient.resolve(), certificate.resolve())
    protect_inputs(args.output, inputs)

    results = []
    binary = args.binary.resolve()
    binary_sha256 = digest(binary)
    for case, paths in pending.items():
        completed = subprocess.run(
            [str(binary), *(str(value) for value in case), *(str(path) for path in paths)],
            check=False, capture_output=True, encoding="utf-8", errors="replace",
            timeout=args.timeout_seconds,
        )
        require(completed.returncode == 0, f"{case} exit {completed.returncode}: {completed.stderr}")
        report = json.loads(completed.stdout)
        verify_result(report, case, certificates[case])
        report["coefficient_sha256"] = certificates[case]["input_sha256"]
        report["certificate_sha256"] = certificates[case]["sha256"]
        results.append(report)
        print(f"{case}: rank {report['rank']}, coordinate comparisons {report['coordinate_comparisons']}",
              flush=True)

    protect_inputs(args.output, inputs)
    atomic_write(args.output, {
        "schema": "uvlas.pg-rank-bounded-product-series/1",
        "index_sha256": INDEX_SHA256,
        "native_binary_sha256": binary_sha256,
        "product_plan_cap_bytes": RANK_CAP,
        "rank_step_cap": STEP_CAP,
        "rank_digit_cap": DIGIT_CAP,
        "timeout_seconds_per_case": args.timeout_seconds,
        "cases": results,
    })


if __name__ == "__main__":
    main()
