cmake_minimum_required(VERSION 3.25)
project(UVLASNumericalDependencySpike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(UVLAS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
include("${UVLAS_ROOT}/cmake/CompilerPolicy.cmake")
set(DEPS "${UVLAS_ROOT}/build/numerical-deps" CACHE PATH "Pinned header source root")
foreach(name IN ITEMS dense_sparse precision_hash capacity solver_contracts)
  add_executable(${name} ${name}.cpp)
  target_include_directories(${name} SYSTEM PRIVATE "${DEPS}/eigen-5.0.1" "${DEPS}/spectra-1.2.0/include" "${DEPS}/boost_1_92_0")
  target_compile_definitions(${name} PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE "EIGEN_DEFAULT_DENSE_INDEX_TYPE=std::int64_t")
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus /fp:strict)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
  endif()
endforeach()
if(WIN32)
  target_link_libraries(capacity PRIVATE psapi)
  target_compile_definitions(capacity PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
enable_testing()
add_test(NAME dense_sparse COMMAND dense_sparse)
add_test(NAME precision_hash COMMAND precision_hash)
add_test(NAME actual_j1_matrix COMMAND dense_sparse "${UVLAS_ROOT}/build/analysis/numeric-oracles/pg-free-step.txt")
add_test(NAME solver_contracts COMMAND solver_contracts)
