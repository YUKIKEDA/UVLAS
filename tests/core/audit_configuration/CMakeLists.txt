cmake_minimum_required(VERSION 3.25)
project(AllocationAuditConfiguration LANGUAGES CXX)
include("${AUDIT_MODULE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/code.cpp" "int sample() { return 0; }\n")
if(CASE MATCHES "^object-|^imported-source-")
    add_library(product OBJECT "${CMAKE_CURRENT_BINARY_DIR}/code.cpp")
else()
    add_library(product STATIC "${CMAKE_CURRENT_BINARY_DIR}/code.cpp")
endif()
add_library(audit STATIC "${CMAKE_CURRENT_BINARY_DIR}/code.cpp")
uvlas_enable_allocation_audit(audit)
if(CASE STREQUAL "object-good")
    uvlas_enable_allocation_audit(product)
    target_sources(audit PRIVATE $<TARGET_OBJECTS:product>)
elseif(CASE STREQUAL "object-bad")
    target_sources(audit PRIVATE $<TARGET_OBJECTS:product>)
elseif(CASE MATCHES "^imported-")
    add_library(wrapper INTERFACE IMPORTED)
    if(CASE MATCHES "good$")
        uvlas_enable_allocation_audit(product)
    endif()
    if(CASE MATCHES "^imported-source-")
        set_property(TARGET wrapper PROPERTY INTERFACE_SOURCES "$<TARGET_OBJECTS:product>")
    else()
        set_property(TARGET wrapper PROPERTY INTERFACE_LINK_LIBRARIES product)
    endif()
    target_link_libraries(audit PRIVATE wrapper)
elseif(CASE STREQUAL "raw-source")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/unknown.obj" "opaque")
    target_sources(audit PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/unknown.obj")
elseif(CASE STREQUAL "raw-library")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/unknown.lib" "opaque")
    target_link_libraries(audit PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/unknown.lib")
elseif(CASE STREQUAL "source-expression")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/unknown.obj" "opaque")
    target_sources(audit PRIVATE "$<$<CONFIG:Debug>:${CMAKE_CURRENT_BINARY_DIR}/unknown.obj>")
elseif(CASE STREQUAL "directory-source")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/child")
    set(opaque "${CMAKE_CURRENT_BINARY_DIR}/child/opaque.payload")
    file(WRITE "${opaque}" "opaque")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/child/CMakeLists.txt"
        "add_library(child STATIC \"${CMAKE_CURRENT_BINARY_DIR}/code.cpp\" \"${opaque}\")\nset_source_files_properties(\"${opaque}\" PROPERTIES EXTERNAL_OBJECT TRUE)\nuvlas_enable_allocation_audit(child)\n")
    add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/child" "${CMAKE_CURRENT_BINARY_DIR}/child-build")
    target_link_libraries(audit PRIVATE child)
elseif(CASE STREQUAL "good")
    uvlas_enable_allocation_audit(product)
    target_link_libraries(audit PRIVATE product)
elseif(CASE STREQUAL "hidden")
    add_library(inner INTERFACE)
    add_library(outer INTERFACE)
    target_link_libraries(inner INTERFACE product)
    target_link_libraries(outer INTERFACE inner)
    target_link_libraries(audit PRIVATE outer)
elseif(CASE STREQUAL "bad")
    target_link_libraries(audit PRIVATE product)
else()
    message(FATAL_ERROR "Unknown CASE")
endif()
uvlas_require_audited_dependencies(audit)
