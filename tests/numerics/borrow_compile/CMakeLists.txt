cmake_minimum_required(VERSION 3.25)
project(UvlasStorageCompilation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
find_package(Eigen3 CONFIG REQUIRED)
if(NOT PROJECT_ROOT)
    message(FATAL_ERROR "PROJECT_ROOT is required")
endif()
set(prefix "#include <numerics/eigen_views.hpp>\nusing namespace uvlas::numerics;\nusing uvlas::core::Operation;\nnamespace adapter = uvlas::numerics::detail;\n")
set(control "${prefix}void check(DenseReal& product, Eigen::MatrixXd& foreign) { auto a=adapter::eigen_read_map(product,0,0,Operation::SolveLinear); auto b=adapter::borrow_eigen_storage<double>(foreign,{0,0,0},Operation::SolveLinear); }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/control.cpp" "${control}")
try_compile(valid SOURCES "${CMAKE_CURRENT_BINARY_DIR}/control.cpp" NO_CACHE
    CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${PROJECT_ROOT}/include;${PROJECT_ROOT}/src"
    LINK_LIBRARIES Eigen3::Eigen OUTPUT_VARIABLE output)
if(NOT valid)
    message(FATAL_ERROR "Positive borrow compilation failed: ${output}")
endif()
set(cases unsupported-scalar temporary-product temporary-eigen temporary-points)
set(unsupported-scalar "DenseMatrix<float> unsupported;")
set(temporary-product "void check() { auto view=adapter::eigen_read_map(DenseReal{},0,0,Operation::SolveLinear); }")
set(temporary-eigen "void check() { auto view=adapter::borrow_eigen_storage<double>(Eigen::MatrixXd{}, {0,0,0}, Operation::SolveLinear); }")
set(temporary-points "void check() { auto view=PointSoA{}.x(); }")
foreach(case IN LISTS cases)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${case}.cpp" "${prefix}${${case}}\n")
    try_compile(accepted SOURCES "${CMAKE_CURRENT_BINARY_DIR}/${case}.cpp" NO_CACHE
        CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${PROJECT_ROOT}/include;${PROJECT_ROOT}/src"
        LINK_LIBRARIES Eigen3::Eigen OUTPUT_VARIABLE output)
    if(accepted)
        message(FATAL_ERROR "Forbidden storage usage compiled: ${case}")
    endif()
    message(STATUS "Rejected ${case}")
endforeach()
